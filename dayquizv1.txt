        // REAL-TIME COUNTDOWN TIMER TO MIDNIGHT (12:00 AM)
        function startMidnightCountdown() {
            if (countdownTimerInterval) clearInterval(countdownTimerInterval);

            function updateCountdownUI() {
                const now = new Date();
                const currentDateStr = getTodayLocalString();

                // Check if date switched while user is on page
                if (activeCheckDate && activeCheckDate !== currentDateStr) {
                    activeCheckDate = currentDateStr;
                    clearInterval(countdownTimerInterval);
                    handleMidnightRollover();
                    return;
                }

                const tomorrow = new Date(now.getFullYear(), now.getMonth(), now.getDate() + 1, 0, 0, 0);
                const diffMs = tomorrow - now;

                if (diffMs <= 0) {
                    clearInterval(countdownTimerInterval);
                    handleMidnightRollover();
                    return;
                }

                const hours = Math.floor(diffMs / (1000 * 60 * 60));
                const minutes = Math.floor((diffMs % (1000 * 60 * 60)) / (1000 * 60));
                const seconds = Math.floor((diffMs % (1000 * 60)) / 1000);

                const timeFormatted = `${String(hours).padStart(2, '0')}h ${String(minutes).padStart(2, '0')}m ${String(seconds).padStart(2, '0')}s`;

                const resCountdown = document.getElementById('result-countdown');
                const compCountdown = document.getElementById('completed-countdown');

                if (resCountdown) resCountdown.innerText = timeFormatted;
                if (compCountdown) compCountdown.innerText = timeFormatted;
            }

            updateCountdownUI();
            countdownTimerInterval = setInterval(updateCountdownUI, 1000);
        }

/**
 * GOOGLE APPS SCRIPT BACKEND FOR QUIZ APP
 * Handles dynamic spreadsheet tab loading, strict persistent row pointer batching, and leaderboards.
 */

function doGet(e) {
  autoSetupDatabase();

  var action = e ? e.parameter.action : null;
  var response = {};

  try {
    if (action === "getSubjects") {
      response = { success: true, subjects: getSubjectsList() };
    } else if (action === "getQuestions") {
      var subject = e.parameter.subject || "English";
      var count = parseInt(e.parameter.count, 10) || 10;
      response = { 
        success: true, 
        subject: subject, 
        questions: getQuestionsForSubject(subject, count) 
      };
    } else if (action === "submitScore") {
      var sub = e.parameter.subject || "English";
      var name = e.parameter.name || "Anonymous";
      var score = parseInt(e.parameter.score, 10) || 0;
      var total = parseInt(e.parameter.total, 10) || 5;
      var timeTaken = parseInt(e.parameter.timeTaken, 10) || 0;

      response = {
        success: true,
        subject: sub,
        leaderboard: recordScoreAndGetLeaderboard(sub, name, score, total, timeTaken)
      };
    } else if (action === "getLeaderboard") {
      var subLeader = e.parameter.subject || "English";
      response = {
        success: true,
        subject: subLeader,
        leaderboard: getTodayLeaderboard(subLeader)
      };
    } else {
      response = { 
        success: true, 
        message: "Quiz API active", 
        subjects: getSubjectsList() 
      };
    }
  } catch (err) {
    response = { success: false, error: err.toString() };
  }

  return ContentService.createTextOutput(JSON.stringify(response))
    .setMimeType(ContentService.MimeType.JSON);
}

/**
 * Automatically creates required Subject Tabs and Leaderboard Tab with standard header columns.
 */
function autoSetupDatabase() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  var headers = [
    "Question", 
    "Option A", 
    "Option B", 
    "Option C", 
    "Option D", 
    "Correct option", 
    "Explanation", 
    "Disable Loop"
  ];

  var defaultSubjectNames = ["English", "Telugu", "Hindi", "General Tech"];

  defaultSubjectNames.forEach(function(subjectName) {
    var sheet = ss.getSheetByName(subjectName);
    
    if (!sheet) {
      sheet = ss.insertSheet(subjectName);
    }

    if (sheet.getLastRow() === 0) {
      sheet.appendRow(headers);
      sheet.getRange(1, 1, 1, headers.length)
        .setFontWeight("bold")
        .setBackground("#4c1d95")
        .setFontColor("#ffffff");
      
      sheet.autoResizeColumns(1, headers.length);
    }
  });

  var leaderboardSheet = ss.getSheetByName("Leaderboard");
  if (!leaderboardSheet) {
    leaderboardSheet = ss.insertSheet("Leaderboard");
    leaderboardSheet.appendRow(["Date", "Subject", "Name", "Score", "Total", "TimeTakenSec", "Timestamp"]);
    leaderboardSheet.getRange(1, 1, 1, 7)
      .setFontWeight("bold")
      .setBackground("#1e1b4b")
      .setFontColor("#ffffff");
  }

  var defaultSheet = ss.getSheetByName("Sheet1");
  if (defaultSheet && ss.getSheets().length > 1) {
    try { ss.deleteSheet(defaultSheet); } catch(e){}
  }
}

/**

/**
 * Parses option letters (A, B, C, D) or numbers into 0-indexed integer.
 */
function parseCorrectOption(val) {
  if (val === undefined || val === null) return 0;
  var str = String(val).trim().toUpperCase();
  if (str === "A") return 0;
  if (str === "B") return 1;
  if (str === "C") return 2;
  if (str === "D") return 3;
  
  var num = parseInt(str, 10);
  if (!isNaN(num)) return num > 0 ? num - 1 : num;
  return 0;
}

/**
 * Strictly fetches questions TOP-TO-BOTTOM from Spreadsheet rows.
 * Tracks daily row pointer via ScriptProperties (Day 1: Rows 2-11, Day 2: Rows 12-21, etc.)
 */
function getQuestionsForSubject(subject, count) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(subject);
  
  if (!sheet) {
    throw new Error("Subject tab '" + subject + "' not found.");
  }

  var lastRow = sheet.getLastRow();
  if (lastRow <= 1) return [];

  var batchSize = count || 10;
  var totalDataRows = lastRow - 1;

  var todayStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
  var props = PropertiesService.getScriptProperties();
  
  var propDateKey = "DATE_" + subject;
  var propRowKey = "ROW_" + subject;
  
  var savedDate = props.getProperty(propDateKey);
  var savedRow = parseInt(props.getProperty(propRowKey), 10);

  var startRow = 2; // Default start row after header

  if (savedDate === todayStr && !isNaN(savedRow) && savedRow >= 2) {
    // Same day: reuse saved starting row pointer
    startRow = savedRow;
  } else {
    // New day or initial run: calculate next block of rows
    if (savedDate && !isNaN(savedRow)) {
      startRow = savedRow + batchSize;
      if (startRow > lastRow) {
        startRow = 2; // Loop back to top row
      }
    } else {
      startRow = 2;
    }
    props.setProperty(propDateKey, todayStr);
    props.setProperty(propRowKey, startRow.toString());
  }

  var endRow = Math.min(startRow + batchSize - 1, lastRow);
  var numRowsToFetch = (endRow - startRow) + 1;

  if (numRowsToFetch <= 0) return [];

  var rawRange = sheet.getRange(startRow, 1, numRowsToFetch, 7).getValues();
  var selectedQuestions = [];

  for (var i = 0; i < rawRange.length; i++) {
    var row = rawRange[i];
    if (row[0] && String(row[0]).trim() !== "") {
      selectedQuestions.push({
        id: startRow + i,
        question: String(row[0]).trim(),
        options: [
          row[1] !== undefined ? String(row[1]).trim() : "",
          row[2] !== undefined ? String(row[2]).trim() : "",
          row[3] !== undefined ? String(row[3]).trim() : "",
          row[4] !== undefined ? String(row[4]).trim() : ""
        ],
        correct: parseCorrectOption(row[5]),
        explanation: (row[6] && String(row[6]).trim() !== "") ? String(row[6]).trim() : "No explanation provided."
      });
    }
  }

  // Wrap around to top rows if the tab has fewer remaining rows than batchSize
  if (selectedQuestions.length < batchSize && totalDataRows > selectedQuestions.length) {
    var remainingNeeded = batchSize - selectedQuestions.length;
    var wrapRange = sheet.getRange(2, 1, Math.min(remainingNeeded, lastRow - 1), 7).getValues();
    for (var j = 0; j < wrapRange.length && selectedQuestions.length < batchSize; j++) {
      var wRow = wrapRange[j];
      if (wRow[0] && String(wRow[0]).trim() !== "") {
        selectedQuestions.push({
          id: 2 + j,
          question: String(wRow[0]).trim(),
          options: [
            wRow[1] !== undefined ? String(wRow[1]).trim() : "",
            wRow[2] !== undefined ? String(wRow[2]).trim() : "",
            wRow[3] !== undefined ? String(wRow[3]).trim() : "",
            wRow[4] !== undefined ? String(wRow[4]).trim() : ""
          ],
          correct: parseCorrectOption(wRow[5]),
          explanation: (wRow[6] && String(wRow[6]).trim() !== "") ? String(wRow[6]).trim() : "No explanation provided."
        });
      }
    }
  }

  return selectedQuestions;
}

/**
 * Records score to backend and fetches updated today's leaderboard.
 */
function recordScoreAndGetLeaderboard(subject, name, score, total, timeTaken) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Leaderboard");
  if (!sheet) {
    autoSetupDatabase();
    sheet = ss.getSheetByName("Leaderboard");
  }

  var todayStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
  sheet.appendRow([todayStr, subject, name, score, total, timeTaken, new Date()]);

  return getTodayLeaderboard(subject);
}

/**
 * Returns strictly subject-filtered, sorted day-wise rank list.
 */
function getTodayLeaderboard(subject) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Leaderboard");
  if (!sheet) return [];

  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];

  var todayStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
  var filtered = [];
  var targetSubject = String(subject || "").trim().toLowerCase();

  for (var i = 1; i < data.length; i++) {
    var rowDate = data[i][0];
    if (rowDate instanceof Date) {
      rowDate = Utilities.formatDate(rowDate, Session.getScriptTimeZone(), "yyyy-MM-dd");
    } else {
      rowDate = String(rowDate).trim();
    }

    var rowSubject = String(data[i][1] || "").trim().toLowerCase();

    if (rowDate === todayStr && rowSubject === targetSubject) {
      filtered.push({
        name: data[i][2],
        score: Number(data[i][3]),
        total: Number(data[i][4]),
        timeTaken: Number(data[i][5])
      });
    }
  }

  filtered.sort(function(a, b) {
    if (b.score !== a.score) {
      return b.score - a.score;
    }
    return a.timeTaken - b.timeTaken;
  });

  return filtered.map(function(item, idx) {
    var mins = Math.floor(item.timeTaken / 60);
    var secs = item.timeTaken % 60;
    var timeFormatted = (mins > 0 ? mins + "m " : "") + secs + "s";
    return {
      rank: idx + 1,
      name: item.name,
      score: item.score,
      total: item.total,
      timeTakenFormatted: timeFormatted,
      timeTakenSeconds: item.timeTaken
    };
  });
}

        function getTodayLocalString() {
            const now = new Date();
            const yyyy = now.getFullYear();
            const mm = String(now.getMonth() + 1).padStart(2, '0');
            const dd = String(now.getDate()).padStart(2, '0');
            return `${yyyy}-${mm}-${dd}`;

        function startMidnightCountdown() {
            if (countdownTimerInterval) clearInterval(countdownTimerInterval);

            function updateCountdownUI() {
                const now = new Date();
                const currentDateStr = getTodayLocalString();

                if (activeCheckDate && activeCheckDate !== currentDateStr) {
                    activeCheckDate = currentDateStr;
                    clearInterval(countdownTimerInterval);
                    handleMidnightRollover();
                    return;
                }

                const tomorrow = new Date(now.getFullYear(), now.getMonth(), now.getDate() + 1, 0, 0, 0);
                const diffMs = tomorrow - now;

                if (diffMs <= 0) {
                    clearInterval(countdownTimerInterval);
                    handleMidnightRollover();
                    return;
                }

                const hours = Math.floor(diffMs / (1000 * 60 * 60));
                const minutes = Math.floor((diffMs % (1000 * 60 * 60)) / (1000 * 60));
                const seconds = Math.floor((diffMs % (1000 * 60)) / 1000);

                const timeFormatted = `${String(hours).padStart(2, '0')}h ${String(minutes).padStart(2, '0')}m ${String(seconds).padStart(2, '0')}s`;

                const resCountdown = document.getElementById('result-countdown');
                const compCountdown = document.getElementById('completed-countdown');

                if (resCountdown) resCountdown.innerText = timeFormatted;
                if (compCountdown) compCountdown.innerText = timeFormatted;
            }

            updateCountdownUI();
            countdownTimerInterval = setInterval(updateCountdownUI, 1000);
        }

        async function fetchBackendSubjects() {
            try {
                const response = await fetch(`${APPS_SCRIPT_URL}?action=getSubjects`);
                const data = await response.json();

                if (data.success && data.subjects.length > 0) {
                    backendSubjects = data.subjects;
                    backendStatus.innerText = "Connected";

                    const urlParams = new URLSearchParams(window.location.search);
                    const sharedSub = urlParams.get('subject');

                    if (sharedSub && backendSubjects.some(s => s.toLowerCase() === sharedSub.toLowerCase())) {
                        const matchedSubject = backendSubjects.find(s => s.toLowerCase() === sharedSub.toLowerCase());
                        currentSubject = matchedSubject;
                    } else {
                        const uncompleted = backendSubjects.find(s => !isSubjectCompletedToday(s));
                        currentSubject = uncompleted || backendSubjects[0];
                    }

                    renderSubjectChips();
                    selectSubject(currentSubject);
                } else {
                    backendStatus.innerText = "No Subjects Found";
                    questionTitle.innerText = "No subjects available in backend spreadsheet.";
                }
            } catch (err) {
                console.error("Backend offline or URL not configured yet:", err);
                backendStatus.innerText = "Connection Failed";
                questionTitle.innerText = "Unable to connect to backend server.";
            }
        }

        function renderSubjectChips() {
            if (backendSubjects.length === 0) return;
            chipsTrack.innerHTML = '';

            const loopSubjects = backendSubjects.length < 5 
                ? [...backendSubjects, ...backendSubjects, ...backendSubjects] 
                : [...backendSubjects, ...backendSubjects];

            loopSubjects.forEach((subject) => {
                const chip = document.createElement('div');
                const isCompleted = isSubjectCompletedToday(subject);
                const isActive = subject === currentSubject;

                chip.className = `subject-chip ${isActive ? 'active' : ''} ${isCompleted ? 'completed' : ''}`;
                chip.innerHTML = `${isCompleted ? '✔ ' : ''}${subject}`;
                chip.onclick = (e) => {
                    e.stopPropagation();
                    selectSubject(subject);
                };

                chipsTrack.appendChild(chip);
            });
        }

        function setupInfiniteScrollEngine() {
            const scrollSpeed = 0.5;

            function scrollStep() {
                if (!isUserInteracting && chipsContainer && chipsTrack.scrollWidth > 0) {
                    chipsContainer.scrollLeft += scrollSpeed;
                    const halfWidth = chipsTrack.scrollWidth / 2;
                    if (chipsContainer.scrollLeft >= halfWidth) {
                        chipsContainer.scrollLeft -= halfWidth;
                    }
                }
                requestAnimationFrame(scrollStep);
            }

            requestAnimationFrame(scrollStep);

            chipsContainer.addEventListener('mouseenter', () => isUserInteracting = true);
            chipsContainer.addEventListener('mouseleave', () => isUserInteracting = false);
            chipsContainer.addEventListener('touchstart', () => isUserInteracting = true, { passive: true });
            chipsContainer.addEventListener('touchend', () => {
                setTimeout(() => isUserInteracting = false, 2000);
            });
        }

        async function selectSubject(subject) {
            if (!subject) return;
            currentSubject = subject;

            closeConceptView();

            todayLeaderboard = [];
            renderLeaderboardUI();

            fetchLeaderboard(currentSubject);

            alreadyCompletedScreen.style.display = 'none';

            quizView.style.display = 'block';
            if (countdownTimerInterval) clearInterval(countdownTimerInterval);

            if (loadingInterval) clearInterval(loadingInterval);
            let step = 0;
            const messages = [

            try {

                const data = await response.json();

                if (loadingInterval) clearInterval(loadingInterval);

                if (data.success && data.questions.length > 0) {
                    quizQuestions = data.questions;
                    userAnswers = new Array(quizQuestions.length).fill(null);
                    currentQuestion = 0;
                    quizStartTime = Date.now();
                    refreshAd();
                    loadQuestion();
                } else {
                    questionTitle.innerText = `No questions added yet under "${subject}".`;
                    optionsGrid.innerHTML = '';
                }
            } catch (err) {
                if (loadingInterval) clearInterval(loadingInterval);
                console.error("Error fetching subject questions:", err);
                questionTitle.innerText = "Failed to load questions from backend.";
                optionsGrid.innerHTML = '';
            }
        }

        function loadQuestion() {
            if (!quizStartTime) quizStartTime = Date.now();

            if (currentQuestion > 0 && currentQuestion % 2 === 0) {
                refreshAd();
            }

            const q = quizQuestions[currentQuestion];
            const savedAns = userAnswers[currentQuestion];

            currentSubjectTitle.innerText = `Subject: ${currentSubject}`;
            tracker.innerText = `Question ${currentQuestion + 1}/${quizQuestions.length}`;
            questionTitle.innerText = q.question;
            progressBar.style.width = `${((currentQuestion + 1) / quizQuestions.length) * 100}%`;

            optionsGrid.innerHTML = '';

            const prefixes = ['A', 'B', 'C', 'D'];

            q.options.forEach((opt, idx) => {
                const chip = document.createElement('div');
                chip.className = 'option-chip';
                chip.innerHTML = `
                    <div class="option-content">
                        <span class="option-prefix">${prefixes[idx]}</span>
                        <span>${opt}</span>
                    </div>
                `;
                chip.onclick = () => handleOptionClick(idx);
                optionsGrid.appendChild(chip);
            });

            if (savedAns !== null) {
                applySelectionUI(savedAns);
            }

            updateNavButtons();
        }

        function handleOptionClick(index) {
            if (!getUserName()) {
                promptUserName(() => selectOption(index));
                return;
            }
            selectOption(index);
        }

        function selectOption(index) {
            if (userAnswers[currentQuestion] !== null) return;

            userAnswers[currentQuestion] = index;
            applySelectionUI(index);
            updateNavButtons();
        }

        function applySelectionUI(selectedIndex) {
            const chips = optionsGrid.querySelectorAll('.option-chip');

            chips.forEach((chip, idx) => {
                chip.classList.add('disabled');
                if (idx === selectedIndex) {
                    chip.classList.add('selected');
                } else {
                    chip.classList.remove('selected');
                }
            });
        }

        function updateNavButtons() {
            const isAnswered = userAnswers[currentQuestion] !== null;

            prevBtn.disabled = currentQuestion === 0;

            if (currentQuestion === quizQuestions.length - 1) {
                nextBtn.innerText = "Submit Quiz";
                nextBtn.className = "btn btn-submit";
            } else {
                nextBtn.innerText = "Next Question";
                nextBtn.className = "btn btn-primary";
            }

            nextBtn.disabled = !isAnswered;
        }

        function navigateQuestion(step) {
            if (step === 1 && currentQuestion === quizQuestions.length - 1) {
                finishQuiz();
                return;
            }

            currentQuestion += step;
            loadQuestion();
        }

        async function finishQuiz() {
            setSubjectCompletedToday(currentSubject);
            renderSubjectChips();

            let score = 0;
            userAnswers.forEach((ans, idx) => {
                if (ans === quizQuestions[idx].correct) score++;
            });

            const totalTimeSec = Math.max(1, Math.floor((Date.now() - quizStartTime) / 1000));
            const userName = getUserName() || "Participant";

            quizView.style.display = 'none';
            resultScreen.style.display = 'block';
            document.getElementById('score-text').innerText = `${score}/${quizQuestions.length}`;

            const badgeEl = document.getElementById('result-badge');
            const titleEl = document.getElementById('result-title');
            const descEl = document.getElementById('result-desc');

            if (score === quizQuestions.length) {
                badgeEl.innerText = "🏆";
                titleEl.innerText = "Perfect Score! Mastermind!";
                descEl.innerText = "Outstanding! You answered every question correctly.";
            } else if (score >= Math.ceil(quizQuestions.length * 0.7)) {
                badgeEl.innerText = "🌟";
                titleEl.innerText = "Great Try! Fantastic Score!";
                descEl.innerText = "Awesome work! You possess great knowledge in this subject.";
            } else {
                badgeEl.innerText = "💪";
                titleEl.innerText = "Good Effort!";
                descEl.innerText = "Keep practicing! Review explanations and try another subject.";
            }

            renderReviewSection();
            startMidnightCountdown();
            resultScreen.scrollTop = 0;

            await submitScoreToBackend(userName, score, quizQuestions.length, totalTimeSec);
        }

        function renderReviewSection() {
            const reviewBox = document.getElementById('review-container');
            if (!reviewBox) return;

            reviewBox.innerHTML = '<h3 class="review-header-title">📋 Detailed Question Review</h3>';

function doGet(e) {
  autoSetupDatabase();

  var action = e ? e.parameter.action : null;
  var response = {};

  try {
    if (action === "getSubjects") {
      response = { success: true, subjects: getSubjectsList() };
    } else if (action === "getQuestions") {
      var subject = e.parameter.subject || "English";
      var count = parseInt(e.parameter.count, 10) || 10;
      response = { 
        success: true, 
        subject: subject, 
        questions: getQuestionsForSubject(subject, count) 
      };
    } else if (action === "submitScore") {
      var sub = e.parameter.subject || "English";
      var name = e.parameter.name || "Anonymous";
      var score = parseInt(e.parameter.score, 10) || 0;
      var total = parseInt(e.parameter.total, 10) || 5;
      var timeTaken = parseInt(e.parameter.timeTaken, 10) || 0;

      response = {
        success: true,
        subject: sub,
        leaderboard: recordScoreAndGetLeaderboard(sub, name, score, total, timeTaken)
      };
    } else if (action === "getLeaderboard") {
      var subLeader = e.parameter.subject || "English";
      response = {
        success: true,
        subject: subLeader,
        leaderboard: getTodayLeaderboard(subLeader)
      };
    } else if (action === "getConceptFolders") {
      response = {
        success: true,
        folders: getConceptFoldersTree()
      };
    } else {
      response = { 
        success: true, 
        message: "Quiz API active", 
        subjects: getSubjectsList(),
        folders: getConceptFoldersTree()
      };
    }
  } catch (err) {
    response = { success: false, error: err.toString() };
  }

  return ContentService.createTextOutput(JSON.stringify(response))
    .setMimeType(ContentService.MimeType.JSON);
}

function parseCorrectOption(val) {
  if (val === undefined || val === null) return 0;
  var str = String(val).trim().toUpperCase();
  if (str === "A") return 0;
  if (str === "B") return 1;
  if (str === "C") return 2;
  if (str === "D") return 3;

  var num = parseInt(str, 10);
  if (!isNaN(num)) return num > 0 ? num - 1 : num;
  return 0;
}

/**
 * Strictly fetches questions TOP-TO-BOTTOM from Spreadsheet rows.
 * Tracks daily row pointer via ScriptProperties (Day 1: Rows 2-11, Day 2: Rows 12-21, etc.)
 */
function getQuestionsForSubject(subject, count) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(subject);

  if (!sheet) {
    throw new Error("Subject tab '" + subject + "' not found.");
  }

  var lastRow = sheet.getLastRow();
  if (lastRow <= 1) return [];

  var batchSize = count || 10;
  var totalDataRows = lastRow - 1;

  var todayStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
  var props = PropertiesService.getScriptProperties();

  var propDateKey = "DATE_" + subject;
  var propRowKey = "ROW_" + subject;

  var savedDate = props.getProperty(propDateKey);
  var savedRow = parseInt(props.getProperty(propRowKey), 10);

  var startRow = 2; // Default start row after header

  if (savedDate === todayStr && !isNaN(savedRow) && savedRow >= 2) {
    // Same day: reuse saved starting row pointer
    startRow = savedRow;
  } else {
    // New day or initial run: calculate next block of rows
    if (savedDate && !isNaN(savedRow)) {
      startRow = savedRow + batchSize;
      if (startRow > lastRow) {
        startRow = 2; // Loop back to top row
      }
    } else {
      startRow = 2;
    }
    props.setProperty(propDateKey, todayStr);
    props.setProperty(propRowKey, startRow.toString());
  }

  var endRow = Math.min(startRow + batchSize - 1, lastRow);
  var numRowsToFetch = (endRow - startRow) + 1;

  if (numRowsToFetch <= 0) return [];

  var rawRange = sheet.getRange(startRow, 1, numRowsToFetch, 7).getValues();
  var selectedQuestions = [];

  for (var i = 0; i < rawRange.length; i++) {
    var row = rawRange[i];
    if (row[0] && String(row[0]).trim() !== "") {
      selectedQuestions.push({
        id: startRow + i,
        question: String(row[0]).trim(),
        options: [
          row[1] !== undefined ? String(row[1]).trim() : "",
          row[2] !== undefined ? String(row[2]).trim() : "",
          row[3] !== undefined ? String(row[3]).trim() : "",
          row[4] !== undefined ? String(row[4]).trim() : ""
        ],
        correct: parseCorrectOption(row[5]),
        explanation: (row[6] && String(row[6]).trim() !== "") ? String(row[6]).trim() : "No explanation provided."
      });
    }
  }

  // Wrap around to top rows if the tab has fewer remaining rows than batchSize
  if (selectedQuestions.length < batchSize && totalDataRows > selectedQuestions.length) {
    var remainingNeeded = batchSize - selectedQuestions.length;
    var wrapRange = sheet.getRange(2, 1, Math.min(remainingNeeded, lastRow - 1), 7).getValues();
    for (var j = 0; j < wrapRange.length && selectedQuestions.length < batchSize; j++) {
      var wRow = wrapRange[j];
      if (wRow[0] && String(wRow[0]).trim() !== "") {
        selectedQuestions.push({
          id: 2 + j,
          question: String(wRow[0]).trim(),
          options: [
            wRow[1] !== undefined ? String(wRow[1]).trim() : "",
            wRow[2] !== undefined ? String(wRow[2]).trim() : "",
            wRow[3] !== undefined ? String(wRow[3]).trim() : "",
            wRow[4] !== undefined ? String(wRow[4]).trim() : ""
          ],
          correct: parseCorrectOption(wRow[5]),
          explanation: (wRow[6] && String(wRow[6]).trim() !== "") ? String(wRow[6]).trim() : "No explanation provided."
        });
      }
    }
  }

  return selectedQuestions;
}

function getTodayLeaderboard(subject) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Leaderboard");
  if (!sheet) return [];

  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];

  var todayStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
  var filtered = [];
  var targetSubject = String(subject || "").trim().toLowerCase();

  for (var i = 1; i < data.length; i++) {
    var rowDate = data[i][0];
    if (rowDate instanceof Date) {
      rowDate = Utilities.formatDate(rowDate, Session.getScriptTimeZone(), "yyyy-MM-dd");
    } else {
      rowDate = String(rowDate).trim();
    }

    var rowSubject = String(data[i][1] || "").trim().toLowerCase();

    if (rowDate === todayStr && rowSubject === targetSubject) {
      filtered.push({
        name: data[i][2],
        score: Number(data[i][3]),
        total: Number(data[i][4]),
        timeTaken: Number(data[i][5])
      });
    }
  }

  filtered.sort(function(a, b) {
    if (b.score !== a.score) {
      return b.score - a.score;
    }
    return a.timeTaken - b.timeTaken;
  });

  return filtered.map(function(item, idx) {
    var mins = Math.floor(item.timeTaken / 60);
    var secs = item.timeTaken % 60;
    var timeFormatted = (mins > 0 ? mins + "m " : "") + secs + "s";
    return {
      rank: idx + 1,
      name: item.name,
      score: item.score,
      total: item.total,
      timeTakenFormatted: timeFormatted,
      timeTakenSeconds: item.timeTaken
    };
  });
}

function getConceptFoldersTree() {
  var sheet = getSheet("ConceptFolders");
  if (!sheet) return [];

  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];

  var items = [];
  for (var i = 1; i < data.length; i++) {
    var r = data[i];
    if (r[0] !== undefined && r[0] !== "") {
      var itemType = String(r[3] || "FOLDER").trim().toLowerCase();
      items.push({
        id: String(r[0]),
        parentId: String(r[1] || "0"),
        name: String(r[2] || ""),
        type: (itemType === "link" || itemType === "test") ? "test" : "folder",
        link: String(r[4] || "")
      });
    }
  }

  function buildTree(pId) {
    var branch = [];
    items.forEach(function(item) {
      if (String(item.parentId) === String(pId)) {
        var children = buildTree(item.id);
        var node = {
          id: item.id,
          parentId: item.parentId,
          name: item.name,
          type: item.type,
          link: item.link,
          children: children
        };
        branch.push(node);
      }
    });
    return branch;
  }

  return buildTree("0");
}

    .indent { margin-left: 22px; border-left: 2px solid #a855f7; padding-left: 10px; margin-top: 6px; }
    button { cursor: pointer; padding: 6px 12px; margin-right: 4px; border-radius: 6px; border: none; font-size: 12px; font-weight: 600; transition: all 0.2s; }
    button:hover { opacity: 0.9; transform: translateY(-1px); }
    .btn-blue { background-color: #2563eb; color: white; }

    .btn-yellow { background-color: #d97706; color: white; }
    .type-badge { font-size: 10px; text-transform: uppercase; padding: 2px 6px; border-radius: 4px; font-weight: bold; margin-right: 6px; }

    .badge-test { background: #dbeafe; color: #1d4ed8; }

<body>
  <div class="toolbar">
    <h3 style="margin:0 0 8px 0; font-size:16px; color:#4c1d95;">📂 Concept Folder Manager</h3>
    <div id="clipboardInfo" class="clipboard-status"></div>
    <div style="display: flex; gap: 6px; flex-wrap: wrap;">
      <button class="btn-green" onclick="addItem('0')">➕ Add Root</button>
      <button class="btn-purple" onclick="save()">💾 Save Changes</button>
      <button class="btn-gray" onclick="google.script.host.close()">❌ Close</button>
    </div>
  </div>

  <div id="container" style="margin-top: 12px;">Loading folder structure...</div>

  <script>
    let items = [];
    let clipboard = null;

    google.script.run.withSuccessHandler(data => {
      items = (data && data.length) ? data : [];
      render();
    }).getTopicData();

    function render() {
      const container = document.getElementById('container');
      container.innerHTML = '';
      if(items.length === 0) {
        container.innerHTML = '<p style="color:#64748b; font-style:italic;">No folders yet. Click <b>+ Add Root</b> to begin.</p>';
        return;
      }
      renderRecursive('0', container);
      updateClipboardUI();
    }

    function renderRecursive(parentId, container) {
      const children = items.filter(i => String(i.parentId) === String(parentId));
      children.forEach(item => {
        const div = document.createElement('div');
        div.className = 'item';
        
        const isFolder = (item.type === 'FOLDER');
        const badgeClass = isFolder ? 'badge-folder' : 'badge-test';
        const icon = isFolder ? '📁' : '📝';

        let html = \`
          <div style="display:flex; justify-content:space-between; align-items:center;">
            <div>
              <span class="type-badge \${badgeClass}">\${item.type}</span>
              <b style="font-size:13px;">\${icon} \${item.name}</b>
              \${!isFolder ? '<br><small style="color:#2563eb; font-weight:600;">Link: ' + (item.url || 'None') + '</small>' : ''}
              <small style="color:#94a3b8; margin-left:6px;">(ID: \${item.id})</small>
            </div>
          </div>
          <div class="actions">
            \${isFolder ? \`<button class="btn-green" title="Add Inside" onclick="addItem('\${item.id}')">➕ Add</button>\` : ''}
            <button class="btn-blue" title="Rename" onclick="renameItem('\${item.id}')">✏️ Edit</button>
            <button class="btn-yellow" title="Cut" onclick="cutItem('\${item.id}')">✂️ Cut</button>
            <button class="btn-gray" title="Copy" onclick="copyItem('\${item.id}')">📋 Copy</button>
        \`;

        if (clipboard && isFolder) {
          html += \`<button class="btn-green" title="Paste Inside" onclick="pasteItem('\${item.id}')">📌 Paste</button>\`;
        }

        html += \`<button class="btn-red" title="Delete" onclick="delItem('\${item.id}')">🗑️ Delete</button></div>\`;
        div.innerHTML = html;
        container.appendChild(div);

        if (isFolder) {
          const subContainer = document.createElement('div');
          subContainer.className = 'indent';
          div.appendChild(subContainer);
          renderRecursive(item.id, subContainer);
        }
      });
    }

    function addItem(parentId) {
      const isFolder = confirm("Click OK for FOLDER \\nClick CANCEL for TEST LINK");
      const type = isFolder ? 'FOLDER' : 'TEST';
      const name = prompt("Enter Name:");
      if (!name) return;

      let url = "";
      if (!isFolder) {
        url = prompt("Enter Test Link (URL or Subject Name e.g., 'English'):");
        if (!url) return;
      }

      const newId = String(Date.now());
      items.push({ id: newId, parentId: String(parentId), name: name, type: type, url: url });
      render();
    }

    function renameItem(id) {
      const item = items.find(i => String(i.id) === String(id));
      if (!item) return;

      const newName = prompt("Rename:", item.name);
      if (newName) {
        item.name = newName;
        if (item.type !== 'FOLDER') {
          const newUrl = prompt("Update Link/Subject:", item.url);
          if (newUrl !== null) item.url = newUrl;
        }
        render();
      }
    }

    function delItem(id) {
      if(confirm("Delete this item and all sub-items inside it?")) {
        const idsToDelete = [String(id)];
        let found = true;
        while(found) {
          found = false;
          items.forEach(i => {
            if(idsToDelete.includes(String(i.parentId)) && !idsToDelete.includes(String(i.id))) {
              idsToDelete.push(String(i.id));
              found = true;
            }
          });
        }
        items = items.filter(i => !idsToDelete.includes(String(i.id)));
        render();
      }
    }

    function cutItem(id) { clipboard = { id: String(id), mode: 'CUT' }; render(); }
    function copyItem(id) { clipboard = { id: String(id), mode: 'COPY' }; render(); }

    function pasteItem(newParentId) {
      if (!clipboard) return;
      if (clipboard.mode === 'CUT') {
        const item = items.find(i => String(i.id) === String(clipboard.id));
        if (item) item.parentId = String(newParentId);
      } else {
        cloneRecursive(clipboard.id, newParentId);
      }
      clipboard = null;
      render();
    }

    function cloneRecursive(sourceId, newParentId) {
      const source = items.find(i => String(i.id) === String(sourceId));
      if(!source) return;
      const newId = String(Date.now() + Math.floor(Math.random() * 1000));
      items.push({
        id: newId,
        parentId: String(newParentId),
        name: source.name + (String(sourceId) === String(clipboard.id) ? " (Copy)" : ""),
        type: source.type,
        url: source.url
      });
      items.filter(i => String(i.parentId) === String(source.id))
           .forEach(child => cloneRecursive(child.id, newId));
    }

    function updateClipboardUI() {
      const div = document.getElementById('clipboardInfo');
      if (clipboard) {
        div.style.display = 'block';
        div.innerHTML = \`<b>\${clipboard.mode === 'CUT' ? '✂️ Moving Item' : '📋 Copying Item'}</b><br>Now click <b>📌 Paste</b> on any target folder.\`;
      } else {
        div.style.display = 'none';
      }
    }

    function save() {
      google.script.run.withSuccessHandler(() => {
        alert("✅ Concept Folders saved successfully!");
      }).saveTopicData(items);
    }
  </script>
</body>
</html>
`;
