        // =========================================================================
        // 12-HOUR (AM/PM) & 24-HOUR TIME CONVERSION UTILITIES
        // =========================================================================
        let use12HourFormat = true;

        function convert24To12Parts(time24) {
            if (!time24 || !/^\d{1,2}:\d{2}$/.test(time24)) {
                return { hour12: "08", minute: "30", ampm: "AM" };
            }
            let [h, m] = time24.split(':').map(Number);
            const ampm = h >= 12 ? "PM" : "AM";
            let h12 = h % 12;
            if (h12 === 0) h12 = 12;
            return {
                hour12: String(h12).padStart(2, '0'),
                minute: String(m).padStart(2, '0'),
                ampm: ampm
            };
        }

        function convert12PartsTo24(hour12Str, minuteStr, ampm) {
            let h = parseInt(hour12Str, 10) || 12;
            let m = parseInt(minuteStr, 10) || 0;
            h = Math.max(1, Math.min(12, h));
            m = Math.max(0, Math.min(59, m));

            if (ampm === "PM" && h < 12) h += 12;
            if (ampm === "AM" && h === 12) h = 0;

            return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}`;
        }

            if (!use12HourFormat) {
                const [h, m] = time24.split(':');
                return `${String(parseInt(h, 10)).padStart(2, '0')}:${m}`;
            }
            const parts = convert24To12Parts(time24);
            return `${parts.hour12}:${parts.minute} ${parts.ampm}`;
        }

        function formatMinutesOfDayForDisplay(totalMins) {
            const safeMins = ((totalMins % 1440) + 1440) % 1440;
            const h24 = String(Math.floor(safeMins / 60)).padStart(2, '0');
            const m = String(safeMins % 60).padStart(2, '0');
            return formatTimeForDisplay(`${h24}:${m}`);
        }

        function parse24TimeToMinutes(time24) {
            if (!time24 || !/^\d{1,2}:\d{2}$/.test(time24)) return null;
            const [h, m] = time24.split(':').map(Number);
            if (isNaN(h) || isNaN(m)) return null;
            return h * 60 + m;
        }

        function ensureColumn24Times(col) {
            if (col.start24 && col.end24) return;
            if (!col.timeLabel) {
                col.start24 = "08:30";
                col.end24 = "09:15";
                return;
            }
            const parts = col.timeLabel.split('-').map(s => s.trim());
            col.start24 = parseFlexibleTimeTokenTo24(parts[0]) || "08:30";
            col.end24 = parseFlexibleTimeTokenTo24(parts[1]) || "09:15";
        }

        function parseFlexibleTimeTokenTo24(token) {
            if (!token) return null;
            const match = token.match(/(\d{1,2}):(\d{2})\s*(am|pm)?/i);
            if (!match) return null;
            let h = parseInt(match[1], 10);
            const m = parseInt(match[2], 10);
            const mer = match[3] ? match[3].toUpperCase() : null;
            if (mer === 'PM' && h < 12) h += 12;
            else if (mer === 'AM' && h === 12) h = 0;
            else if (!mer && h >= 1 && h <= 6) h += 12;
            return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}`;
        }

        function getColumnFormattedRange(col) {
            ensureColumn24Times(col);
            return `${formatTimeForDisplay(col.start24)} - ${formatTimeForDisplay(col.end24)}`;
        }

        // DATE UTILITY FUNCTIONS
        function getTodayDateStr() {
            const now = new Date();
            const y = now.getFullYear();
            const m = String(now.getMonth() + 1).padStart(2, '0');
            const d = String(now.getDate()).padStart(2, '0');
            return `${y}-${m}-${d}`;
        }

        function parseLocalDate(dateStr) {
            const parts = dateStr.split('-');
            if (parts.length === 3) {
                return new Date(parseInt(parts[0], 10), parseInt(parts[1], 10) - 1, parseInt(parts[2], 10));
            }
            return new Date();
        }

        function formatLocalDate(dateObj) {
            const y = dateObj.getFullYear();
            const m = String(dateObj.getMonth() + 1).padStart(2, '0');
            const d = String(dateObj.getDate()).padStart(2, '0');
            return `${y}-${m}-${d}`;
        }

        function calculateDurationMinsFromTimeRange(timeFrom, timeTo, fallbackMins = 45) {
            if (!timeFrom || !timeTo) return fallbackMins;
            const [h1, m1] = timeFrom.split(':').map(Number);
            const [h2, m2] = timeTo.split(':').map(Number);
            if (isNaN(h1) || isNaN(m1) || isNaN(h2) || isNaN(m2)) return fallbackMins;
            let diff = (h2 * 60 + m2) - (h1 * 60 + m1);
            if (diff < 0) diff += 12 * 60;
            return diff > 0 ? diff : fallbackMins;
        }

        // High-precision 1-second Web Worker + main thread fallback so alarms trigger reliably
        function startBackgroundTicker() {
            if (alarmWorker) return;
            try {
                const workerBlob = new Blob([`
                    setInterval(() => { postMessage('tick'); }, 1000);
                `], { type: 'application/javascript' });
                alarmWorker = new Worker(URL.createObjectURL(workerBlob));
                alarmWorker.onmessage = () => {
                    checkPeriodAlarms();
                };
            } catch (e) {
                setInterval(() => {
                    checkPeriodAlarms();
                }, 1000);
            }

            // Refresh scheduled list status badges every 15s
            setInterval(() => {
                renderScheduledAlarmsList();
                updateNextUpcomingClassPreview();
            }, 15000);
        }

        // Keeps WebView / Mobile audio pipeline alive when screen turns off or device locks
        function startBackgroundAudioKeepAlive() {
            try {
                const AudioCtx = window.AudioContext || window.webkitAudioContext;
                if (!keepAliveAudioCtx) {
                    keepAliveAudioCtx = new AudioCtx();
                }
                if (keepAliveAudioCtx.state === 'suspended') {
                    keepAliveAudioCtx.resume();
                }
                if (!keepAliveOsc) {
                    const osc = keepAliveAudioCtx.createOscillator();
                    const gain = keepAliveAudioCtx.createGain();
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(20, keepAliveAudioCtx.currentTime); // Sub-audible 20Hz carrier
                    gain.gain.setValueAtTime(0.0001, keepAliveAudioCtx.currentTime);
                    osc.connect(gain);
                    gain.connect(keepAliveAudioCtx.destination);
                    osc.start();
                    keepAliveOsc = osc;
                }

                if ('mediaSession' in navigator) {
                    navigator.mediaSession.metadata = new MediaMetadata({
                        title: "Teacher's Day Book Alarm Guard",
                        artist: profileData.name || "Class Schedule Active",
                        album: profileData.school || "Offline Logbook"
                    });
                    navigator.mediaSession.playbackState = "playing";
                }
                document.getElementById('alarm-keepalive-badge')?.classList.remove('hidden');
            } catch (e) {}
        }

        function resumeAudioContextIfSuspended() {
            try {
                if (keepAliveAudioCtx && keepAliveAudioCtx.state === 'suspended') {
                    keepAliveAudioCtx.resume();
                }
            } catch (e) {}
        }

        function stopBackgroundAudioKeepAlive() {
            try {
                if (keepAliveOsc) {
                    keepAliveOsc.stop();
                    keepAliveOsc.disconnect();
                    keepAliveOsc = null;
                }
                if ('mediaSession' in navigator) {
                    navigator.mediaSession.playbackState = "none";
                }
                document.getElementById('alarm-keepalive-badge')?.classList.add('hidden');
            } catch (e) {}
        }

        async function requestScreenWakeLock() {
            try {
                if ('wakeLock' in navigator && !wakeLockSentinel) {
                    wakeLockSentinel = await navigator.wakeLock.request('screen');
                    wakeLockSentinel.addEventListener('release', () => {
                        wakeLockSentinel = null;
                    });
                }
            } catch (err) {}
        }

        function releaseScreenWakeLock() {
            if (wakeLockSentinel) {
                wakeLockSentinel.release().catch(() => {});
                wakeLockSentinel = null;
            }
        }

        // Syncs exact scheduled alarm timestamps with Android/Capacitor/Cordova APK bridges if embedded
        function syncNativeOfflineAlarmsBridge() {
            try {
                const slots = getAllTodayAlarmItems();
                const payload = JSON.stringify({
                    enabled: alarmSettings.enabled,
                    repeatCount: alarmSettings.repeatCount,
                    repeatIntervalSec: alarmSettings.repeatIntervalSec,
                    alarms: slots
                });

                // 1. Custom Android WebView JavascriptInterface hook
                if (window.AndroidAlarmBridge && typeof window.AndroidAlarmBridge.syncAlarms === 'function') {
                    window.AndroidAlarmBridge.syncAlarms(payload);
                }

                // 2. Capacitor LocalNotifications offline scheduling hook
                if (window.Capacitor && window.Capacitor.Plugins && window.Capacitor.Plugins.LocalNotifications) {
                    const LocalNotifications = window.Capacitor.Plugins.LocalNotifications;
                    if (alarmSettings.enabled) {
                        const now = new Date();
                        const notifications = slots
                            .filter(s => s.status === 'upcoming')
                            .map((s, idx) => {
                                const triggerDate = new Date(now.getFullYear(), now.getMonth(), now.getDate(), Math.floor(s.triggerMins / 60), s.triggerMins % 60, 0);
                                return {
                                    id: 1000 + idx,
                                    title: s.title,
                                    body: s.voiceMessage,
                                    schedule: { at: triggerDate, allowWhileIdle: true }
                                };
                            });
                        if (notifications.length > 0) {
                            LocalNotifications.schedule({ notifications }).catch(() => {});
                        }
                    }
                }
            } catch (e) {}
        }

        // UNIFIED FULL-DAY SCHEDULED ALARMS BUILDER (CLASS PERIODS + PERSONAL ALARMS)
        function getAllTodayAlarmItems() {
            const now = new Date();
            const currentMins = now.getHours() * 60 + now.getMinutes();
            const dayName = now.toLocaleDateString('en-US', { weekday: 'long' });
            const offset = parseInt(alarmSettings.offsetMins, 10) || 0;

            const items = [];

            // 1. Scheduled Class Periods from Today's Timetable
            const targetRow = timetableData.rows.find(r => r.dayName.toLowerCase() === dayName.toLowerCase());
            if (targetRow) {
                timetableData.columns.forEach(col => {
                    if (col.isBreak) return;
                    const cell = timetableData.grid[`${targetRow.id}_${col.id}`] || {};
                    if (!cell.className && !cell.subject) return;

                    ensureColumn24Times(col);
                    const startMins = parse24TimeToMinutes(col.start24);
                    if (startMins === null) return;

                    const triggerMins = Math.max(0, startMins - offset);
                    const slotObj = {
                        colId: col.id,
                        periodTitle: col.title || 'Period',
                        timeLabel: getColumnFormattedRange(col),
                        className: cell.className || '',
                        section: cell.section || '',
                        subject: cell.subject || ''
                    };

                    items.push({
                        uniqueId: `class_${col.id}`,
                        type: 'class',
                        title: `${slotObj.periodTitle}: ${slotObj.className} ${slotObj.section} (${slotObj.subject || 'Class'})`,
                        classTimeDisplay: slotObj.timeLabel,
                        triggerMins: triggerMins,
                        triggerTimeDisplay: formatMinutesOfDayForDisplay(triggerMins),
                        status: triggerMins < currentMins ? 'completed' : (triggerMins === currentMins ? 'ringing' : 'upcoming'),
                        voiceMessage: buildVoiceMessage(slotObj, offset)
                    });
                });
            }

            // 2. Enabled Personal Alarms
            personalAlarms.forEach(pa => {
                if (!pa.enabled) return;
                const triggerMins = parse24TimeToMinutes(pa.time24);
                if (triggerMins === null) return;

                items.push({
                    uniqueId: pa.id,
                    type: 'personal',
                    title: `Personal Reminder: ${pa.message}`,
                    classTimeDisplay: formatTimeForDisplay(pa.time24),
                    triggerMins: triggerMins,
                    triggerTimeDisplay: formatTimeForDisplay(pa.time24),
                    status: triggerMins < currentMins ? 'completed' : (triggerMins === currentMins ? 'ringing' : 'upcoming'),
                    voiceMessage: `Attention ${profileData.name || 'Teacher'}. Personal Reminder: ${pa.message}.`
                });
            });

            return items.sort((a, b) => a.triggerMins - b.triggerMins);
        }

        function checkPeriodAlarms() {
            if (!alarmSettings.enabled) return;

            const now = new Date();
            const todayKey = getTodayDateStr();
            const currentMins = now.getHours() * 60 + now.getMinutes();

            const items = getAllTodayAlarmItems();
            items.forEach(item => {
                const uniqueKey = `${todayKey}_${item.uniqueId}_${item.triggerMins}`;
                // Trigger on exact minute (or within 1-minute catch-up window if device just woke from sleep)
                if ((currentMins === item.triggerMins || currentMins === item.triggerMins + 1) && !triggeredAlarmKeys.has(uniqueKey)) {
                    triggeredAlarmKeys.add(uniqueKey);
                    fireRepeatingAlarmAlert(item);
                    renderScheduledAlarmsList();
                }
            });
        }

        function buildVoiceMessage(slot, offsetMins) {
            const timeIntro = offsetMins > 0 
                ? `In ${offsetMins} minute${offsetMins === 1 ? '' : 's'}, ` 
                : `It is now time for `;
            const classPart = slot.className ? `${slot.className}` : 'your scheduled class';
            const secPart = slot.section ? `, ${slot.section}` : '';
            const subjPart = slot.subject ? `, for ${slot.subject}` : '';
            const timePart = slot.timeLabel ? ` at ${slot.timeLabel}` : '';
            return `Attention ${profileData.name || 'Teacher'}. ${timeIntro}${slot.periodTitle}${timePart}. Please go to ${classPart}${secPart}${subjPart}.`;
        }

        function stopActiveRepeatLoop() {
            if (activeRepeatTimer) {
                clearInterval(activeRepeatTimer);
                activeRepeatTimer = null;
            }
            activeRepeatRemaining = 0;
        }

        function fireRepeatingAlarmAlert(item) {
            stopActiveRepeatLoop();

            const message = item.voiceMessage;
            lastSpokenMessage = message;

            const maxRepeats = parseInt(alarmSettings.repeatCount, 10) || 3;
            const intervalMs = (parseInt(alarmSettings.repeatIntervalSec, 10) || 30) * 1000;
            let currentRingNumber = 1;
            activeRepeatRemaining = maxRepeats - 1;

            const banner = document.getElementById('live-alarm-banner');
            const bannerText = document.getElementById('live-alarm-text');
            const badgeText = document.getElementById('live-alarm-badge');
            const counterText = document.getElementById('live-alarm-repeat-counter');

            const triggerSingleRing = () => {
                if (banner && bannerText) {
                    bannerText.innerText = message;
                    if (badgeText) badgeText.innerText = item.type === 'personal' ? 'Personal Reminder' : 'Class Period Alert';
                    if (counterText) {
                        counterText.innerText = maxRepeats === 999
                            ? `Ring #${currentRingNumber} (Repeating)`
                            : `Ring ${currentRingNumber} of ${maxRepeats}`;
                    }
                    banner.classList.remove('hidden');
                }

                // Vibrate mobile device if supported
                if ('vibrate' in navigator) {
                    try { navigator.vibrate([400, 200, 400, 200, 600]); } catch (e) {}
                }

                playAlarmChime();
                setTimeout(() => speakText(message), 950);

                if ('Notification' in window && Notification.permission === 'granted') {
                    try {
                        const notif = new Notification(`🔔 ${item.title}`, {
                            body: message,
                            tag: `alarm-${item.uniqueId}`,
                            renotify: true,
                            requireInteraction: true
                        });
                        notif.onclick = () => {
                            window.focus();
                            notif.close();
                        };
                    } catch (e) {}
                }
            };

            // Ring immediately on Ring #1
            triggerSingleRing();

            // Schedule subsequent repeat reminders based on Repeat Count & Interval Frequency
            if (activeRepeatRemaining > 0) {
                activeRepeatTimer = setInterval(() => {
                    if (activeRepeatRemaining <= 0) {
                        stopActiveRepeatLoop();
                        return;
                    }
                    currentRingNumber++;
                    if (maxRepeats !== 999) {
                        activeRepeatRemaining--;
                    }
                    triggerSingleRing();
                }, intervalMs);
            }
        }

        function playAlarmChime() {
            try {
                const AudioCtx = window.AudioContext || window.webkitAudioContext;
                const ctx = new AudioCtx();
                const notes = [587.33, 783.99, 1046.50, 1318.51];
                notes.forEach((freq, idx) => {
                    const osc = ctx.createOscillator();
                    const gain = ctx.createGain();
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(freq, ctx.currentTime + idx * 0.2);
                    gain.gain.setValueAtTime(0.32, ctx.currentTime + idx * 0.2);
                    gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + idx * 0.2 + 0.65);
                    osc.connect(gain);
                    gain.connect(ctx.destination);
                    osc.start(ctx.currentTime + idx * 0.2);
                    osc.stop(ctx.currentTime + idx * 0.2 + 0.7);
                });
            } catch (e) {}
        }

        function speakText(text) {
            if (!('speechSynthesis' in window)) return;
            window.speechSynthesis.cancel();
            const utterance = new SpeechSynthesisUtterance(text);
            utterance.rate = 0.95;
            utterance.pitch = 1.0;
            utterance.volume = 1.0;
            window.speechSynthesis.speak(utterance);
        }

        function editConfigItem(itemsKey, index) {
            const oldVal = profileData[itemsKey][index];
            const newVal = prompt(`Edit entry name:`, oldVal);
            if (newVal !== null && newVal.trim() !== '') {
                const trimmed = newVal.trim();
                profileData[itemsKey][index] = trimmed;

                Object.keys(appData).forEach(dateKey => {
                    const log = appData[dateKey];
                    (log.periods || []).forEach(p => {
                        if (itemsKey === 'classes' && p.className === oldVal) p.className = trimmed;
                        if (itemsKey === 'sections' && p.section === oldVal) p.section = trimmed;
                        if (itemsKey === 'workTypes' && p.workType === oldVal) p.workType = trimmed;
                    });
                    (log.extraWorks || []).forEach(w => {
                        if (itemsKey === 'classes' && w.className === oldVal) w.className = trimmed;
                        if (itemsKey === 'extraWorkTypes' && w.category === oldVal) w.category = trimmed;
                    });
                });

                Object.keys(timetableData.grid || {}).forEach(cellKey => {
                    const cell = timetableData.grid[cellKey];
                    if (itemsKey === 'classes' && cell.className === oldVal) cell.className = trimmed;
                    if (itemsKey === 'sections' && cell.section === oldVal) cell.section = trimmed;

                renderProfileConfigLists();
                loadDateLog();

                populateSearchDropdowns();
                syncAllDependentViews();
            }

        function handleExtraWorkFileUpload(event, idx) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                currentExtraWorks[idx].attachment = e.target.result;
                currentExtraWorks[idx].fileName = file.name;
                renderExtraWorkList();
                saveLog(false);
                syncAllDependentViews();
                showToast("File attached successfully!");
            };
            reader.readAsDataURL(file);
        }

                syncAllDependentViews();

        }

        // LOAD DATE LOG DATA
        function loadDateLog() {
            const input = document.getElementById('log-date');
            if (input && input.value) currentDateStr = input.value;
            
            const dayLog = appData[currentDateStr] || {};

            if (document.getElementById('homeroom-duty')) document.getElementById('homeroom-duty').value = dayLog.homeroom || '';
            if (document.getElementById('day-notes')) document.getElementById('day-notes').value = dayLog.dayNotes || '';
            if (document.getElementById('next-day-plan')) document.getElementById('next-day-plan').value = dayLog.nextPlan || '';

            currentPeriods = Array.isArray(dayLog.periods) ? JSON.parse(JSON.stringify(dayLog.periods)) : [];
            renderPeriodRows();

            currentExtraWorks = Array.isArray(dayLog.extraWorks) ? JSON.parse(JSON.stringify(dayLog.extraWorks)) : [];
            renderExtraWorkList();
        }

        // SAVE LOG TO LOCAL STORAGE
        function saveLog(showToastNotification = false) {
            const input = document.getElementById('log-date');
            if (input && input.value) currentDateStr = input.value;
            if (!currentDateStr) return;

            const homeroomVal = document.getElementById('homeroom-duty') ? document.getElementById('homeroom-duty').value.trim() : '';
            const dayNotesVal = document.getElementById('day-notes') ? document.getElementById('day-notes').value.trim() : '';
            const nextPlanVal = document.getElementById('next-day-plan') ? document.getElementById('next-day-plan').value.trim() : '';

            const normalizedPeriods = currentPeriods.map((p, idx) => ({
                ...p,
                periodNum: idx + 1,
                periodLabel: p.periodLabel || `Period ${idx + 1}`,
                durationMins: calculateDurationMinsFromTimeRange(p.timeFrom, p.timeTo, 45)
            }));

            if (normalizedPeriods.length === 0 && currentExtraWorks.length === 0 && !homeroomVal && !dayNotesVal && !nextPlanVal) {
                delete appData[currentDateStr];
            } else {
                appData[currentDateStr] = {
                    homeroom: homeroomVal,
                    dayNotes: dayNotesVal,
                    nextPlan: nextPlanVal,
                    periods: normalizedPeriods,
                    extraWorks: JSON.parse(JSON.stringify(currentExtraWorks)),
                    updatedAt: new Date().toISOString()
                };
            }

            saveLocalStorageData();
            calculateAndDisplayDailyHours();

            if (showToastNotification) {
                syncAllDependentViews();
                showToast("Logbook saved successfully!");
            }
        }

        function loadFromTimetableForCurrentDate() {
            const dateObj = parseLocalDate(currentDateStr);
            const dayName = dateObj.toLocaleDateString('en-US', { weekday: 'long' });

            const targetRow = timetableData.rows.find(r => r.dayName.toLowerCase() === dayName.toLowerCase());
            if (!targetRow) {
                showToast(`No timetable entry found for ${dayName}`, "info");
                return;
            }

            if (currentPeriods.length > 0 && !confirm(`Append ${dayName}'s timetable classes to your current log rows?`)) {
                return;
            }

            let addedCount = 0;
            timetableData.columns.forEach((col) => {
                if (col.isBreak) return;
                const cellData = timetableData.grid[`${targetRow.id}_${col.id}`] || {};
                if (cellData.className || cellData.subject) {
                    ensureColumn24Times(col);
                    currentPeriods.push({
                        id: `period_${Date.now()}_${addedCount}`,
                        periodLabel: col.title || `Period ${currentPeriods.length + 1}`,
                        timeFrom: col.start24 || '',
                        timeTo: col.end24 || '',
                        className: cellData.className || '',
                        section: cellData.section || '',
                        workType: cellData.workType || 'Teaching',
                        subject: cellData.subject || '',
                        topic: '',
                        activity: '',
                        status: 'Completed'
                    });
                    addedCount++;
                }
            });

            renderPeriodRows();
            saveLog(false);
            syncAllDependentViews();
            showToast(`Imported ${addedCount} classes from ${dayName}'s timetable!`);
        }

        let autoSaveTimer;
        function autoSave() {
            clearTimeout(autoSaveTimer);
            autoSaveTimer = setTimeout(() => { saveLog(false); }, 600);
        }

        function clearCurrentDay() {
            if (confirm(`Delete ALL period entries, extra duties, attachments, and notes for ${currentDateStr}? This will remove it from Reports, Analytics, and Search.`)) {
                delete appData[currentDateStr];
                currentPeriods = [];
                currentExtraWorks = [];

                syncAllDependentViews();

            }
        }

        function calculateAndDisplayDailyHours() {
            let coreMins = 0;
            let extraMins = 0;

            currentPeriods.forEach(p => {
                if (p.status !== 'Cancelled' && p.status !== 'Free/Planning') {
                    const dur = calculateDurationMinsFromTimeRange(p.timeFrom, p.timeTo, 45);
                    coreMins += dur;
                }
            });

            currentExtraWorks.forEach(work => {
                const dur = parseInt(work.durationMins) || 0;
                if (work.isExtraDuty) extraMins += dur;
                else coreMins += dur;
            });

            const formatHM = (mins) => `${Math.floor(mins / 60)}h ${mins % 60}m`;
            const badgeTotal = document.getElementById('daily-total-hours-badge');
            const badgeExtra = document.getElementById('daily-extra-hours-badge');
            if (badgeTotal) badgeTotal.innerText = formatHM(coreMins + extraMins);
            if (badgeExtra) badgeExtra.innerText = formatHM(extraMins);
        }

        function renderAnalytics() {
            const rangeFilter = document.getElementById('analytics-filter-range').value;
            const dutyFilter = document.getElementById('analytics-filter-duty').value;
            const allDates = Object.keys(appData).sort();
            const todayDate = new Date();

            const filteredDates = allDates.filter(dateStr => {
                const d = parseLocalDate(dateStr);
                if (rangeFilter === 'today') return dateStr === currentDateStr;
                if (rangeFilter === 'this_week') return Math.abs((todayDate - d) / (1000 * 60 * 60 * 24)) <= 7;
                if (rangeFilter === 'this_month') return d.getMonth() === todayDate.getMonth() && d.getFullYear() === todayDate.getFullYear();
                return true;
            });

            let totalCoreMins = 0, totalExtraMins = 0, totalPeriodsCount = 0, totalExtraTasksCount = 0;
            const categoryMinutesMap = {};
            const dailyLogsBreakdown = [];

            filteredDates.forEach(dateStr => {
                const dayLog = appData[dateStr];
                let dayCoreMins = 0, dayExtraMins = 0, dayPeriods = 0;

                if (dutyFilter !== 'extra') {
                    (dayLog.periods || []).forEach((p) => {
                        if (p.status !== 'Cancelled' && p.status !== 'Free/Planning') {
                            const duration = p.durationMins || calculateDurationMinsFromTimeRange(p.timeFrom, p.timeTo, 45);
                            dayCoreMins += duration; dayPeriods++; totalPeriodsCount++;
                            const category = p.workType || 'Teaching';
                            categoryMinutesMap[category] = (categoryMinutesMap[category] || 0) + duration;
                        }
                    });
                }

                if (dutyFilter !== 'core') {
                    (dayLog.extraWorks || []).forEach(work => {
                        const dur = parseInt(work.durationMins) || 0;
                        totalExtraTasksCount++;
                        if (work.isExtraDuty) dayExtraMins += dur; else dayCoreMins += dur;
                        const category = work.category || 'Extra Duty';
                        categoryMinutesMap[category] = (categoryMinutesMap[category] || 0) + dur;
                    });
                }

                totalCoreMins += dayCoreMins; totalExtraMins += dayExtraMins;
                if (dayCoreMins > 0 || dayExtraMins > 0) {
                    dailyLogsBreakdown.push({ date: dateStr, coreMins: dayCoreMins, extraMins: dayExtraMins, totalMins: dayCoreMins + dayExtraMins, periods: dayPeriods });
                }
            });

            const grandTotalMins = totalCoreMins + totalExtraMins;
            const formatHM = (mins) => `${Math.floor(mins / 60)}h ${mins % 60}m`;

            document.getElementById('stat-total-work-hours').innerText = formatHM(grandTotalMins);
            document.getElementById('stat-total-days-count').innerText = `Across ${filteredDates.length} logged days`;
            document.getElementById('stat-core-hours').innerText = formatHM(totalCoreMins);
            document.getElementById('stat-core-periods-count').innerText = `${totalPeriodsCount} Core Periods`;
            document.getElementById('stat-extra-hours').innerText = formatHM(totalExtraMins);
            document.getElementById('stat-extra-tasks-count').innerText = `${totalExtraTasksCount} Extra Duties`;

            const avgDailyMins = filteredDates.length > 0 ? Math.round(grandTotalMins / filteredDates.length) : 0;
            document.getElementById('stat-avg-daily-hours').innerText = formatHM(avgDailyMins);

            const categoryContainer = document.getElementById('worktype-breakdown-list');
            categoryContainer.innerHTML = '';
            const sortedCategories = Object.keys(categoryMinutesMap).sort((a,b) => categoryMinutesMap[b] - categoryMinutesMap[a]);

            if (sortedCategories.length === 0) {
                categoryContainer.innerHTML = `<p class="text-xs text-slate-400 italic">No duty breakdown data available.</p>`;
            } else {
                sortedCategories.forEach(cat => {
                    const mins = categoryMinutesMap[cat];
                    const percent = grandTotalMins > 0 ? Math.round((mins / grandTotalMins) * 100) : 0;
                    const row = document.createElement('div');
                    row.className = "space-y-1.5";
                    row.innerHTML = `
                        <div class="flex justify-between text-xs">
                            <span class="font-bold text-slate-700 flex items-center gap-2"><i class="fa-solid fa-circle text-[8px] text-brand-500"></i> ${cat}</span>
                            <span class="font-semibold text-slate-600">${formatHM(mins)} (${percent}%)</span>
                        </div>
                        <div class="w-full bg-slate-100 rounded-full h-2 overflow-hidden flex">
                            <div class="bg-brand-600 h-full rounded-full" style="width: ${percent}%"></div>
                        </div>
                    `;
                    categoryContainer.appendChild(row);
                });
            }

            const summaryContainer = document.getElementById('daily-hours-summary-list');
            summaryContainer.innerHTML = '';
            if (dailyLogsBreakdown.length === 0) {
                summaryContainer.innerHTML = `<p class="text-xs text-slate-400 italic">No daily logs recorded for this period.</p>`;
            } else {
                dailyLogsBreakdown.sort((a,b) => b.date.localeCompare(a.date)).forEach(item => {
                    const row = document.createElement('div');
                    row.className = "p-3 bg-slate-50 border border-slate-200 rounded-xl flex items-center justify-between hover:bg-slate-100 cursor-pointer";
                    row.onclick = () => jumpToDate(item.date);
                    row.innerHTML = `
                        <div>
                            <span class="text-xs font-bold text-slate-800 block">${item.date}</span>
                            <span class="text-[11px] text-slate-500">${item.periods} Periods</span>
                        </div>
                        <div class="flex items-center gap-3 text-xs">
                            <div class="text-right"><span class="text-[9px] uppercase font-bold text-slate-400 block">Core</span><span class="font-bold text-emerald-700">${formatHM(item.coreMins)}</span></div>
                            <div class="text-right"><span class="text-[9px] uppercase font-bold text-amber-500 block">Extra</span><span class="font-bold text-amber-700">${formatHM(item.extraMins)}</span></div>
                            <div class="text-right border-l pl-3"><span class="text-[9px] uppercase font-bold text-brand-500 block">Total</span><span class="font-black text-brand-700">${formatHM(item.totalMins)}</span></div>
                        </div>
                    `;
                    summaryContainer.appendChild(row);
                });
            }
        }

        // MANAGEMENT REPORT SYSTEM (ACCURATE A4 PAGINATION + FIT + TOUCH PAN & PINCH ZOOM)

        function initA4TouchPanAndPinchZoom() {
            const viewport = document.getElementById('report-a4-viewport');
            if (!viewport) return;

            let initialPinchDistance = null;
            let initialZoomAtPinch = reportZoomPercent;

            let isMousePanning = false;
            let startMouseX = 0, startMouseY = 0;
            let startScrollLeft = 0, startScrollTop = 0;

            viewport.addEventListener('mousedown', (e) => {
                if (e.button !== 0) return;
                isMousePanning = true;
                startMouseX = e.clientX;
                startMouseY = e.clientY;
                startScrollLeft = viewport.scrollLeft;
                startScrollTop = viewport.scrollTop;
            });

            window.addEventListener('mousemove', (e) => {
                if (!isMousePanning) return;
                e.preventDefault();
                viewport.scrollLeft = startScrollLeft - (e.clientX - startMouseX);
                viewport.scrollTop = startScrollTop - (e.clientY - startMouseY);
            });

            window.addEventListener('mouseup', () => {
                isMousePanning = false;
            });

            viewport.addEventListener('touchstart', (e) => {
                if (e.touches.length === 2) {
                    const dx = e.touches[0].clientX - e.touches[1].clientX;
                    const dy = e.touches[0].clientY - e.touches[1].clientY;
                    initialPinchDistance = Math.hypot(dx, dy);
                    initialZoomAtPinch = reportZoomPercent;
                }
            }, { passive: true });

            viewport.addEventListener('touchmove', (e) => {
                if (e.touches.length === 2 && initialPinchDistance) {
                    e.preventDefault();
                    const dx = e.touches[0].clientX - e.touches[1].clientX;
                    const dy = e.touches[0].clientY - e.touches[1].clientY;
                    const currentDistance = Math.hypot(dx, dy);
                    const scaleRatio = currentDistance / initialPinchDistance;
                    const nextZoom = Math.round(Math.max(35, Math.min(180, initialZoomAtPinch * scaleRatio)));
                    if (nextZoom !== reportZoomPercent) {
                        reportZoomPercent = nextZoom;
                        applyReportZoom();
                    }
                }
            }, { passive: false });

            viewport.addEventListener('touchend', (e) => {
                if (e.touches.length < 2) {
                    initialPinchDistance = null;
                }
            }, { passive: true });
        }

        function adjustReportZoom(delta) {

        function fitReportToScreen() {
            const viewport = document.getElementById('report-a4-viewport');
            const stage = document.getElementById('report-a4-zoom-stage');

            const availableWidth = Math.max(240, viewport.clientWidth - 36);
            const naturalWidth = stage.offsetWidth || 794;
            const idealPercent = Math.floor((availableWidth / naturalWidth) * 100);
            reportZoomPercent = Math.max(35, Math.min(115, idealPercent));
            applyReportZoom();
        }

                        </div>
                    `;

                    const periodRows = (log.periods || []).map((p, idx) => {

                            ? `<div class="text-[9px] text-slate-500 font-normal">${formatTimeForDisplay(p.timeFrom)} - ${formatTimeForDisplay(p.timeTo)}</div>`
                            : '';

                            <tr class="border-b border-slate-200 text-[11px]">
                                <td class="py-1.5 px-2.5 font-bold text-slate-700 w-28">${p.periodLabel || `Period ${idx + 1}`}${timeDisplay}</td>

                                <td class="py-1.5 px-2.5 text-slate-700 font-medium w-24">${p.workType || 'Teaching'}</td>
                                <td class="py-1.5 px-2.5 font-bold text-slate-900 w-28">${p.subject || '-'}</td>
                                <td class="py-1.5 px-2.5 text-slate-700">${p.topic || '-'} ${p.activity ? `<span class="text-slate-500 font-normal">(${p.activity})</span>` : ''}</td>

                            <td class="py-1.5 px-2.5 font-bold text-amber-800"><i class="fa-solid fa-user-clock text-amber-600 mr-1"></i> Extra Duty</td>
                            <td class="py-1.5 px-2.5 font-semibold text-amber-900">${w.className || 'General'}</td>
                            <td class="py-1.5 px-2.5 text-amber-800 font-bold">${w.category || 'Special Duty'}</td>

        function populateSearchDropdowns() {
            const classSelect = document.getElementById('search-class');
            const worktypeSelect = document.getElementById('search-worktype');
            if (classSelect) classSelect.innerHTML = '<option value="">All Classes</option>' + profileData.classes.map(c => `<option value="${c}">${c}</option>`).join('');

        function deletePeriodFromSearch(dateStr, periodIndex) {
            if (!appData[dateStr] || !appData[dateStr].periods) return;
            if (!confirm(`Delete this period entry from ${dateStr} across all logs, analytics, and reports?`)) return;

            appData[dateStr].periods.splice(periodIndex, 1);
            const log = appData[dateStr];
            if ((log.periods || []).length === 0 && (log.extraWorks || []).length === 0 && !log.homeroom && !log.dayNotes && !log.nextPlan) {
                delete appData[dateStr];
            }
            saveLocalStorageData();
            if (dateStr === currentDateStr) loadDateLog();
            syncAllDependentViews();
            showToast("Period entry deleted everywhere.", "info");
        }

        function deleteExtraWorkFromSearch(dateStr, extraIndex) {
            if (!appData[dateStr] || !appData[dateStr].extraWorks) return;
            if (!confirm(`Delete this extra duty entry from ${dateStr} across all logs, analytics, and reports?`)) return;

            appData[dateStr].extraWorks.splice(extraIndex, 1);
            const log = appData[dateStr];
            if ((log.periods || []).length === 0 && (log.extraWorks || []).length === 0 && !log.homeroom && !log.dayNotes && !log.nextPlan) {
                delete appData[dateStr];
            }
            saveLocalStorageData();
            if (dateStr === currentDateStr) loadDateLog();
            syncAllDependentViews();
            showToast("Extra work entry deleted everywhere.", "info");
        }

        function executeSearch() {
            const keyword = document.getElementById('search-keyword').value.toLowerCase().trim();
            const classFilter = document.getElementById('search-class').value.toLowerCase().trim();
            const worktypeFilter = document.getElementById('search-worktype').value.toLowerCase().trim();
            const container = document.getElementById('search-results-list');
            if (!container) return;
            container.innerHTML = '';

            let matchCount = 0;
            Object.keys(appData).sort().reverse().forEach(date => {
                const log = appData[date];

                const matchingPeriods = (log.periods || []).map((p, origIdx) => ({ p, origIdx })).filter(({ p }) => {
                    const matchKwd = !keyword || (p.topic && p.topic.toLowerCase().includes(keyword)) || (p.activity && p.activity.toLowerCase().includes(keyword)) || (p.subject && p.subject.toLowerCase().includes(keyword));
                    const matchClass = !classFilter || (p.className && p.className.toLowerCase().includes(classFilter));
                    const matchWorktype = !worktypeFilter || (p.workType && p.workType.toLowerCase().includes(worktypeFilter));
                    return matchKwd && matchClass && matchWorktype;
                });

                const matchingExtraWorks = (log.extraWorks || []).map((w, origIdx) => ({ w, origIdx })).filter(({ w }) => {
                    const matchKwd = !keyword || (w.description && w.description.toLowerCase().includes(keyword));
                    const matchClass = !classFilter || (w.className && w.className.toLowerCase().includes(classFilter));
                    const matchWorktype = !worktypeFilter || (w.category && w.category.toLowerCase().includes(worktypeFilter));
                    return matchKwd && matchClass && matchWorktype;
                });

                if (matchingPeriods.length > 0 || matchingExtraWorks.length > 0) {
                    matchCount += (matchingPeriods.length + matchingExtraWorks.length);
                    const dateHeader = document.createElement('div');
                    dateHeader.className = "p-3 bg-slate-100 font-bold text-xs text-slate-700 flex justify-between items-center";
                    dateHeader.innerHTML = `<span><i class="fa-regular fa-calendar-days text-brand-600 mr-2"></i> ${date}</span><button onclick="jumpToDate('${date}')" class="text-brand-600 hover:underline no-print">Open Log &rrarr;</button>`;
                    container.appendChild(dateHeader);

                    matchingPeriods.forEach(({ p, origIdx }) => {
                        const item = document.createElement('div');
                        item.className = "p-3 hover:bg-slate-50 border-b border-slate-100 text-xs flex items-start justify-between gap-3";
                        const timeStr = (p.timeFrom && p.timeTo) ? ` [${formatTimeForDisplay(p.timeFrom)} - ${formatTimeForDisplay(p.timeTo)}]` : '';
                        item.innerHTML = `
                            <div>
                                <p class="font-bold text-slate-800">${p.periodLabel || `Period ${origIdx + 1}`}${timeStr}: ${p.subject || 'Class'} (${p.className || 'General'} ${p.section || ''}) - <span class="text-brand-600">${p.workType || 'Teaching'}</span></p>
                                <p class="text-slate-600">Topic: ${p.topic || 'N/A'}</p>
                            </div>
                            <button onclick="deletePeriodFromSearch('${date}', ${origIdx})" title="Delete Entry Everywhere" class="text-slate-400 hover:text-rose-600 p-1.5 rounded-lg hover:bg-rose-50 transition no-print shrink-0">
                                <i class="fa-solid fa-trash-can"></i>
                            </button>
                        `;
                        container.appendChild(item);
                    });

                    matchingExtraWorks.forEach(({ w, origIdx }) => {
                        const item = document.createElement('div');
                        item.className = "p-3 bg-amber-50/40 hover:bg-amber-50 border-b border-amber-100 text-xs space-y-2";
                        item.innerHTML = `
                            <div class="flex items-start justify-between gap-3">
                                <div>
                                    <p class="font-bold text-amber-900"><i class="fa-solid fa-user-clock mr-1"></i> Extra Work: ${w.category} (${w.durationMins} Mins)</p>
                                    <p class="text-amber-800">${w.description || '-'}</p>
                                </div>
                                <button onclick="deleteExtraWorkFromSearch('${date}', ${origIdx})" title="Delete Extra Duty Everywhere" class="text-amber-600 hover:text-rose-600 p-1.5 rounded-lg hover:bg-rose-50 transition no-print shrink-0">
                                    <i class="fa-solid fa-trash-can"></i>
                                </button>
                            </div>
                            ${w.attachment ? `<div class="mt-2"><img src="${w.attachment}" class="max-h-64 rounded-md border border-amber-200" /></div>` : ''}
                        `;
                        container.appendChild(item);
                    });
                }
            });
            document.getElementById('search-count').innerText = `${matchCount} records found`;
        }

        function exportDataCSV() {
            let csv = "data:text/csv;charset=utf-8,Date,Type,Period,TimeFrom,TimeTo,Class,Section,WorkType,Subject,Topic,DurationMins\n";
            Object.keys(appData).sort().forEach(date => {
                const log = appData[date];
                (log.periods || []).forEach((p, idx) => {
                    const dur = p.durationMins || calculateDurationMinsFromTimeRange(p.timeFrom, p.timeTo, 45);
                    csv += `${date},Core,"${p.periodLabel || `Period ${idx + 1}`}","${formatTimeForDisplay(p.timeFrom)}","${formatTimeForDisplay(p.timeTo)}","${p.className || ''}","${p.section || ''}","${p.workType || ''}","${p.subject || ''}","${p.topic || ''}",${dur}\n`;
                });
                (log.extraWorks || []).forEach(w => {
                    csv += `${date},Extra,N/A,,,"${w.className || ''}",,"${w.category || ''}","Extra Duty","${w.description || ''}",${w.durationMins || 0}\n`;
                });
            });
            const link = document.createElement('a');
            link.setAttribute('href', encodeURI(csv));
            link.setAttribute('download', `Teacher_Daybook_${getTodayDateStr()}.csv`);
            document.body.appendChild(link); link.click(); link.remove();
        }

        function importDataJSON(e) {
            const file = e.target.files[0];
            if (!file) return;
            const r = new FileReader();
            r.onload = function(evt) {
                try {
                    const imported = JSON.parse(evt.target.result);
                    if (imported.appData) appData = imported.appData;
                    if (imported.profileData) profileData = imported.profileData;
                    if (imported.timetableData) timetableData = imported.timetableData;
                    if (imported.alarmSettings) alarmSettings = imported.alarmSettings;
                    if (imported.personalAlarms) personalAlarms = imported.personalAlarms;
                    use12HourFormat = alarmSettings.use12HourFormat !== false;
                    syncTimeFormatToggleUI();
                    saveLocalStorageData();
                    updateProfileUIHeadings();
                    renderProfileConfigLists(); renderPeriodRows(); loadDateLog(); renderTimetableBuilder(); renderPersonalAlarmTimeInputs(); renderPersonalAlarmsList(); populateSearchDropdowns(); onReportScopeTypeChange(); initAlarmSystem();
                    closeModal('modal-backup');
                    showToast("Backup imported successfully!");
                } catch (err) { alert("Invalid JSON backup file."); }
            };
            r.readAsText(file);
        }

        function switchTab(tabId) {
            currentTabId = tabId;
            ['planner', 'timetable', 'analytics', 'report', 'search'].forEach(t => {
                const view = document.getElementById(`view-${t}`);
                if (view) {
                    view.classList.add('hidden');
                    view.classList.remove('active-print-view');
                }
                const btn = document.getElementById(`tab-${t}`);
                if (btn) { btn.classList.remove('bg-brand-800', 'text-white'); btn.classList.add('text-brand-100'); }
                const mobBtn = document.getElementById(`mob-tab-${t}`);
                if (mobBtn) { mobBtn.classList.remove('text-white'); mobBtn.classList.add('text-brand-200'); }
            });

            const target = document.getElementById(`view-${tabId}`);
            if (target) {
                target.classList.remove('hidden');
                target.classList.add('active-print-view');
            }

            const activeBtn = document.getElementById(`tab-${tabId}`);
            if (activeBtn) { activeBtn.classList.add('bg-brand-800', 'text-white'); activeBtn.classList.remove('text-brand-100'); }

            const activeMobBtn = document.getElementById(`mob-tab-${tabId}`);
            if (activeMobBtn) { activeMobBtn.classList.add('text-white'); activeMobBtn.classList.remove('text-brand-200'); }

            if (tabId === 'timetable') {
                renderTimetableBuilder();
                renderPersonalAlarmTimeInputs();
                renderPersonalAlarmsList();
                renderScheduledAlarmsList();
            }
            if (tabId === 'analytics') renderAnalytics();
            if (tabId === 'report') { 
                onReportScopeTypeChange(); 
                setTimeout(fitReportToScreen, 40);
            }
            if (tabId === 'search') executeSearch();
        }

        function showToast(msg) {

            document.getElementById('toast-msg').innerText = msg;
            toast.classList.remove('translate-y-20', 'opacity-0');

            setTimeout(() => {
                toast.classList.remove('translate-y-0', 'opacity-100');
                toast.classList.add('translate-y-20', 'opacity-0');
            }, 2500);
        }
