        function getTodayDateStr() {
            const now = new Date();
            const y = now.getFullYear();
            const m = String(now.getMonth() + 1).padStart(2, '0');
            const d = String(now.getDate()).padStart(2, '0');
            return `${y}-${m}-${d}`;
        }

        function parseLocalDate(dateStr) {
            const parts = dateStr.split('-');
            if (parts.length === 3) {
                return new Date(parseInt(parts[0], 10), parseInt(parts[1], 10) - 1, parseInt(parts[2], 10));
            }
            return new Date();
        }

        function handleExtraWorkFileUpload(event, idx) {
            const file = event.target.files[0];

                currentExtraWorks[idx].fileName = file.name;
                renderExtraWorkList();
                autoSave();
                showToast("File attached successfully!");
            };
            reader.readAsDataURL(file);

            const dayLog = appData[currentDateStr] || {};

            if (document.getElementById('homeroom-duty')) document.getElementById('homeroom-duty').value = dayLog.homeroom || '';
            if (document.getElementById('day-notes')) document.getElementById('day-notes').value = dayLog.dayNotes || '';
            if (document.getElementById('next-day-plan')) document.getElementById('next-day-plan').value = dayLog.nextPlan || '';

            const periods = dayLog.periods || [];
            const numPeriods = timetableData.columns ? timetableData.columns.length : 8;

            for (let i = 1; i <= numPeriods; i++) {
                const p = periods[i - 1] || {};
                if (document.getElementById(`p${i}-class`)) document.getElementById(`p${i}-class`).value = p.className || '';
                if (document.getElementById(`p${i}-section`)) document.getElementById(`p${i}-section`).value = p.section || '';

                if (document.getElementById(`p${i}-topic`)) document.getElementById(`p${i}-topic`).value = p.topic || '';
                if (document.getElementById(`p${i}-activity`)) document.getElementById(`p${i}-activity`).value = p.activity || '';
                if (document.getElementById(`p${i}-status`)) document.getElementById(`p${i}-status`).value = p.status || 'Completed';
            }

            // Reload isolated extra works array for this date only
            currentExtraWorks = Array.isArray(dayLog.extraWorks) ? JSON.parse(JSON.stringify(dayLog.extraWorks)) : [];
            renderExtraWorkList();
        }

        // SAVE LOG TO LOCAL STORAGE
        function saveLog(showToastNotification = false) {
            const input = document.getElementById('log-date');
            if (input && input.value) currentDateStr = input.value;
            if (!currentDateStr) return;

            const periods = [];
            const numPeriods = timetableData.columns ? timetableData.columns.length : 8;

            for (let i = 1; i <= numPeriods; i++) {
                periods.push({
                    periodNum: i,
                    className: document.getElementById(`p${i}-class`) ? document.getElementById(`p${i}-class`).value : '',
                    section: document.getElementById(`p${i}-section`) ? document.getElementById(`p${i}-section`).value : '',
                    workType: document.getElementById(`p${i}-worktype`) ? document.getElementById(`p${i}-worktype`).value : '',
                    subject: document.getElementById(`p${i}-subject`) ? document.getElementById(`p${i}-subject`).value : '',
                    topic: document.getElementById(`p${i}-topic`) ? document.getElementById(`p${i}-topic`).value : '',
                    activity: document.getElementById(`p${i}-activity`) ? document.getElementById(`p${i}-activity`).value : '',
                    status: document.getElementById(`p${i}-status`) ? document.getElementById(`p${i}-status`).value : 'Completed'
                });
            }

            appData[currentDateStr] = {
                homeroom: document.getElementById('homeroom-duty') ? document.getElementById('homeroom-duty').value.trim() : '',
                dayNotes: document.getElementById('day-notes') ? document.getElementById('day-notes').value.trim() : '',
                nextPlan: document.getElementById('next-day-plan') ? document.getElementById('next-day-plan').value.trim() : '',
                periods: periods,
                extraWorks: JSON.parse(JSON.stringify(currentExtraWorks)),
                updatedAt: new Date().toISOString()
            };

            saveLocalStorageData();
            calculateAndDisplayDailyHours();

            if (showToastNotification) showToast("Logbook saved successfully!");
        }

        function loadFromTimetableForCurrentDate() {
            const dateObj = parseLocalDate(currentDateStr);
            const dayName = dateObj.toLocaleDateString('en-US', { weekday: 'long' });

            const targetRow = timetableData.rows.find(r => r.dayName.toLowerCase() === dayName.toLowerCase());
            if (!targetRow) {
                showToast(`No timetable entry found for ${dayName}`, "info");
                return;
            }

            timetableData.columns.forEach((col, idx) => {
                const i = idx + 1;
                const cellData = timetableData.grid[`${targetRow.id}_${col.id}`] || {};
                
                if (col.isBreak) {
                    if (document.getElementById(`p${i}-status`)) document.getElementById(`p${i}-status`).value = 'Free/Planning';
                } else if (cellData.className || cellData.subject) {
                    if (document.getElementById(`p${i}-class`)) document.getElementById(`p${i}-class`).value = cellData.className || '';
                    if (document.getElementById(`p${i}-section`)) document.getElementById(`p${i}-section`).value = cellData.section || '';
                    if (document.getElementById(`p${i}-worktype`)) document.getElementById(`p${i}-worktype`).value = cellData.workType || '';
                    if (document.getElementById(`p${i}-subject`)) document.getElementById(`p${i}-subject`).value = cellData.subject || '';
                }
            });

            saveLog(false);
            showToast(`Timetable auto-filled for ${dayName}!`);
        }

        function calculateAndDisplayDailyHours() {
            let coreMins = 0;
            let extraMins = 0;
            const numPeriods = timetableData.columns ? timetableData.columns.length : 8;

            for (let i = 1; i <= numPeriods; i++) {
                const status = document.getElementById(`p${i}-status`) ? document.getElementById(`p${i}-status`).value : '';
                const className = document.getElementById(`p${i}-class`) ? document.getElementById(`p${i}-class`).value : '';
                const subject = document.getElementById(`p${i}-subject`) ? document.getElementById(`p${i}-subject`).value : '';

                if ((className || subject) && status !== 'Cancelled' && status !== 'Free/Planning') {
                    const colDuration = (timetableData.columns[i-1] && timetableData.columns[i-1].durationMins) ? timetableData.columns[i-1].durationMins : 45;
                    coreMins += colDuration;
                }
            }

            currentExtraWorks.forEach(work => {
                const dur = parseInt(work.durationMins) || 0;
                if (work.isExtraDuty) extraMins += dur;
                else coreMins += dur;
            });

            const formatHM = (mins) => `${Math.floor(mins / 60)}h ${mins % 60}m`;
            const badgeTotal = document.getElementById('daily-total-hours-badge');
            const badgeExtra = document.getElementById('daily-extra-hours-badge');
            if (badgeTotal) badgeTotal.innerText = formatHM(coreMins + extraMins);
            if (badgeExtra) badgeExtra.innerText = formatHM(extraMins);
        }

        function renderAnalytics() {
            const rangeFilter = document.getElementById('analytics-filter-range').value;
            const dutyFilter = document.getElementById('analytics-filter-duty').value;
            const allDates = Object.keys(appData).sort();
            const todayDate = new Date();

            const filteredDates = allDates.filter(dateStr => {
                const d = parseLocalDate(dateStr);
                if (rangeFilter === 'today') return dateStr === currentDateStr;
                if (rangeFilter === 'this_week') return Math.abs((todayDate - d) / (1000 * 60 * 60 * 24)) <= 7;
                if (rangeFilter === 'this_month') return d.getMonth() === todayDate.getMonth() && d.getFullYear() === todayDate.getFullYear();
                return true;
            });

            let totalCoreMins = 0, totalExtraMins = 0, totalPeriodsCount = 0, totalExtraTasksCount = 0;
            const categoryMinutesMap = {};
            const dailyLogsBreakdown = [];

            filteredDates.forEach(dateStr => {
                const dayLog = appData[dateStr];
                let dayCoreMins = 0, dayExtraMins = 0, dayPeriods = 0;

                if (dutyFilter !== 'extra') {
                    (dayLog.periods || []).forEach((p, idx) => {
                        if ((p.className || p.subject) && p.status !== 'Cancelled' && p.status !== 'Free/Planning') {
                            const duration = (timetableData.columns[idx] && timetableData.columns[idx].durationMins) ? timetableData.columns[idx].durationMins : 45;
                            dayCoreMins += duration; dayPeriods++; totalPeriodsCount++;
                            const category = p.workType || 'Teaching';
                            categoryMinutesMap[category] = (categoryMinutesMap[category] || 0) + duration;
                        }
                    });
                }

                if (dutyFilter !== 'core') {
                    (dayLog.extraWorks || []).forEach(work => {
                        const dur = parseInt(work.durationMins) || 0;
                        totalExtraTasksCount++;
                        if (work.isExtraDuty) dayExtraMins += dur; else dayCoreMins += dur;
                        const category = work.category || 'Extra Duty';
                        categoryMinutesMap[category] = (categoryMinutesMap[category] || 0) + dur;
                    });
                }

                totalCoreMins += dayCoreMins; totalExtraMins += dayExtraMins;
                if (dayCoreMins > 0 || dayExtraMins > 0) {
                    dailyLogsBreakdown.push({ date: dateStr, coreMins: dayCoreMins, extraMins: dayExtraMins, totalMins: dayCoreMins + dayExtraMins, periods: dayPeriods });
                }
            });

            const grandTotalMins = totalCoreMins + totalExtraMins;
            const formatHM = (mins) => `${Math.floor(mins / 60)}h ${mins % 60}m`;

            document.getElementById('stat-total-work-hours').innerText = formatHM(grandTotalMins);
            document.getElementById('stat-total-days-count').innerText = `Across ${filteredDates.length} logged days`;
            document.getElementById('stat-core-hours').innerText = formatHM(totalCoreMins);
            document.getElementById('stat-core-periods-count').innerText = `${totalPeriodsCount} Core Periods`;
            document.getElementById('stat-extra-hours').innerText = formatHM(totalExtraMins);
            document.getElementById('stat-extra-tasks-count').innerText = `${totalExtraTasksCount} Extra Duties`;

            const avgDailyMins = filteredDates.length > 0 ? Math.round(grandTotalMins / filteredDates.length) : 0;
            document.getElementById('stat-avg-daily-hours').innerText = formatHM(avgDailyMins);

            const categoryContainer = document.getElementById('worktype-breakdown-list');
            categoryContainer.innerHTML = '';
            const sortedCategories = Object.keys(categoryMinutesMap).sort((a,b) => categoryMinutesMap[b] - categoryMinutesMap[a]);

            if (sortedCategories.length === 0) {
                categoryContainer.innerHTML = `<p class="text-xs text-slate-400 italic">No duty breakdown data available.</p>`;
            } else {
                sortedCategories.forEach(cat => {
                    const mins = categoryMinutesMap[cat];
                    const percent = grandTotalMins > 0 ? Math.round((mins / grandTotalMins) * 100) : 0;
                    const row = document.createElement('div');
                    row.className = "space-y-1.5";
                    row.innerHTML = `
                        <div class="flex justify-between text-xs">
                            <span class="font-bold text-slate-700 flex items-center gap-2"><i class="fa-solid fa-circle text-[8px] text-brand-500"></i> ${cat}</span>
                            <span class="font-semibold text-slate-600">${formatHM(mins)} (${percent}%)</span>
                        </div>
                        <div class="w-full bg-slate-100 rounded-full h-2 overflow-hidden flex">
                            <div class="bg-brand-600 h-full rounded-full" style="width: ${percent}%"></div>
                        </div>
                    `;
                    categoryContainer.appendChild(row);
                });
            }

            const summaryContainer = document.getElementById('daily-hours-summary-list');
            summaryContainer.innerHTML = '';
            if (dailyLogsBreakdown.length === 0) {
                summaryContainer.innerHTML = `<p class="text-xs text-slate-400 italic">No daily logs recorded for this period.</p>`;
            } else {
                dailyLogsBreakdown.sort((a,b) => b.date.localeCompare(a.date)).forEach(item => {
                    const row = document.createElement('div');
                    row.className = "p-3 bg-slate-50 border border-slate-200 rounded-xl flex items-center justify-between hover:bg-slate-100 cursor-pointer";
                    row.onclick = () => jumpToDate(item.date);
                    row.innerHTML = `
                        <div>
                            <span class="text-xs font-bold text-slate-800 block">${item.date}</span>
                            <span class="text-[11px] text-slate-500">${item.periods} Periods</span>
                        </div>
                        <div class="flex items-center gap-3 text-xs">
                            <div class="text-right"><span class="text-[9px] uppercase font-bold text-slate-400 block">Core</span><span class="font-bold text-emerald-700">${formatHM(item.coreMins)}</span></div>
                            <div class="text-right"><span class="text-[9px] uppercase font-bold text-amber-500 block">Extra</span><span class="font-bold text-amber-700">${formatHM(item.extraMins)}</span></div>
                            <div class="text-right border-l pl-3"><span class="text-[9px] uppercase font-bold text-brand-500 block">Total</span><span class="font-black text-brand-700">${formatHM(item.totalMins)}</span></div>
                        </div>
                    `;
                    summaryContainer.appendChild(row);
                });
            }
        }

        function onReportScopeTypeChange() {
            const scopeType = document.getElementById('report-scope-type').value;
            const container = document.getElementById('report-scope-value-container');
            const select = document.getElementById('report-scope-value');
            const label = document.getElementById('report-scope-value-label');

            if (!container || !select) return;

            if (scopeType === 'complete') {
                container.classList.add('hidden');
                generateManagementReport();
                return;
            }

            container.classList.remove('hidden');
            select.innerHTML = '';
            const allDates = Object.keys(appData).sort().reverse();

            if (scopeType === 'month') {
                label.innerText = 'Select Month';
                const months = new Set();
                allDates.forEach(d => months.add(d.substring(0, 7)));
                if (months.size === 0) months.add(getTodayDateStr().substring(0, 7));
                
                Array.from(months).forEach(mKey => {
                    const [year, month] = mKey.split('-');
                    const mName = new Date(year, parseInt(month) - 1, 1).toLocaleString('default', { month: 'long', year: 'numeric' });
                    const opt = document.createElement('option');
                    opt.value = mKey;
                    opt.innerText = mName;
                    select.appendChild(opt);
                });
            } else if (scopeType === 'day') {
                label.innerText = 'Select Single Day';
                if (allDates.length === 0) allDates.push(getTodayDateStr());
                allDates.forEach(dStr => {
                    const opt = document.createElement('option');
                    opt.value = dStr;
                    opt.innerText = dStr;
                    select.appendChild(opt);
                });
            } else if (scopeType === 'week') {
                label.innerText = 'Select Week Ending Date';
                if (allDates.length === 0) allDates.push(getTodayDateStr());
                allDates.forEach(dStr => {
                    const opt = document.createElement('option');
                    opt.value = dStr;
                    opt.innerText = `Week up to ${dStr}`;
                    select.appendChild(opt);
                });
            } else if (scopeType === 'year') {
                label.innerText = 'Select Academic Year';
                const years = new Set();
                allDates.forEach(d => years.add(d.substring(0, 4)));
                if (years.size === 0) years.add(getTodayDateStr().substring(0, 4));
                Array.from(years).forEach(y => {
                    const opt = document.createElement('option');
                    opt.value = y;
                    opt.innerText = `Year ${y}`;
                    select.appendChild(opt);
                });
            }

            generateManagementReport();
        }

        function generateManagementReport() {
            const wrapper = document.getElementById('report-document-wrapper');
            if (!wrapper) return;
            wrapper.innerHTML = '';

            const scopeType = document.getElementById('report-scope-type').value;
            const scopeVal = document.getElementById('report-scope-value') ? document.getElementById('report-scope-value').value : '';
            const formatHM = (mins) => `${Math.floor(mins / 60)}h ${mins % 60}m`;

            const allDates = Object.keys(appData).sort();

            const filteredDates = allDates.filter(dateStr => {
                if (scopeType === 'complete') return true;
                if (scopeType === 'day') return dateStr === scopeVal;
                if (scopeType === 'month') return dateStr.startsWith(scopeVal);
                if (scopeType === 'year') return dateStr.startsWith(scopeVal);
                if (scopeType === 'week') {
                    const endDt = parseLocalDate(scopeVal);
                    const currDt = parseLocalDate(dateStr);
                    const diffDays = (endDt - currDt) / (1000 * 60 * 60 * 24);
                    return diffDays >= 0 && diffDays < 7;
                }
                return true;
            });

            if (filteredDates.length === 0) {
                wrapper.innerHTML = `
                    <div class="p-12 text-center text-slate-400">
                        <i class="fa-solid fa-file-excel text-5xl mb-3"></i>
                        <p class="text-base font-semibold">No Log Data Found</p>
                        <p class="text-xs">No daily logs or extra work entries recorded for the selected range.</p>
                    </div>
                `;
                return;
            }

            let grandCoreMins = 0;
            let grandExtraMins = 0;

            filteredDates.forEach(d => {
                const log = appData[d];
                (log.periods || []).forEach((p, idx) => {
                    if ((p.className || p.subject) && p.status !== 'Cancelled' && p.status !== 'Free/Planning') {
                        const dur = (timetableData.columns[idx] && timetableData.columns[idx].durationMins) ? timetableData.columns[idx].durationMins : 45;
                        grandCoreMins += dur;
                    }
                });
                (log.extraWorks || []).forEach(w => {
                    const dur = parseInt(w.durationMins) || 0;
                    if (w.isExtraDuty) grandExtraMins += dur; else grandCoreMins += dur;
                });
            });

            // Scope Heading String
            let scopeTitleStr = "COMPLETE ACADEMIC LOGBOOK REPORT";
            if (scopeType === 'month') scopeTitleStr = `MONTHLY LOGBOOK REPORT (${scopeVal})`;
            if (scopeType === 'day') scopeTitleStr = `DAILY LOGBOOK REPORT (${scopeVal})`;
            if (scopeType === 'week') scopeTitleStr = `WEEKLY LOGBOOK REPORT (7 Days to ${scopeVal})`;
            if (scopeType === 'year') scopeTitleStr = `ANNUAL ACADEMIC REPORT (${scopeVal})`;

            // Master Header
            const mainHeader = document.createElement('div');
            mainHeader.className = "border-b-2 border-slate-900 pb-5 mb-6";
            mainHeader.innerHTML = `
                <div class="flex justify-between items-start">
                    <div>
                        <h1 class="text-2xl font-black text-slate-900 uppercase tracking-tight">${profileData.school || 'School Name'}</h1>
                        <h2 class="text-sm font-bold text-slate-700">${scopeTitleStr}</h2>
                        <p class="text-xs text-slate-500 mt-1">Teacher: <strong class="text-slate-800">${profileData.name}</strong> | Department: <strong class="text-slate-800">${profileData.dept}</strong></p>
                    </div>
                    <div class="text-right">
                        <div class="bg-slate-100 p-3 rounded-xl border border-slate-300">
                            <span class="text-[10px] uppercase font-extrabold text-slate-500 block">Report Total Hours</span>
                            <span class="text-lg font-black text-slate-900">${formatHM(grandCoreMins + (includeExtraDutiesInPrint ? grandExtraMins : 0))}</span>
                            <span class="text-[10px] font-bold text-emerald-700 block">Core: ${formatHM(grandCoreMins)}</span>
                            ${includeExtraDutiesInPrint ? `<span class="text-[10px] font-bold text-amber-700 block extra-duty-print-stat">Extra Duty: ${formatHM(grandExtraMins)}</span>` : ''}
                        </div>
                    </div>
                </div>
            `;
            wrapper.appendChild(mainHeader);

            const monthsGroup = {};
            filteredDates.forEach(dateStr => {
                const dateObj = parseLocalDate(dateStr);
                const monthName = dateObj.toLocaleString('default', { month: 'long', year: 'numeric' });
                if (!monthsGroup[monthName]) monthsGroup[monthName] = [];
                monthsGroup[monthName].push(dateStr);
            });

            Object.keys(monthsGroup).forEach(monthName => {
                const datesInMonth = monthsGroup[monthName];
                let monthCoreMins = 0;
                let monthExtraMins = 0;

                datesInMonth.forEach(d => {
                    const log = appData[d];
                    (log.periods || []).forEach((p, idx) => {
                        if ((p.className || p.subject) && p.status !== 'Cancelled' && p.status !== 'Free/Planning') {
                            monthCoreMins += (timetableData.columns[idx] && timetableData.columns[idx].durationMins) ? timetableData.columns[idx].durationMins : 45;
                        }
                    });
                    (log.extraWorks || []).forEach(w => {
                        const dur = parseInt(w.durationMins) || 0;
                        if (w.isExtraDuty) monthExtraMins += dur; else monthCoreMins += dur;
                    });
                });

                const monthSection = document.createElement('div');
                monthSection.className = "space-y-5 pt-4 border-t border-slate-300 first:border-none first:pt-0";
                monthSection.innerHTML = `
                    <div class="bg-slate-900 text-white px-4 py-2 rounded-xl flex justify-between items-center uppercase font-bold text-xs">
                        <span><i class="fa-solid fa-calendar text-brand-400 mr-2"></i> MONTH: ${monthName}</span>
                        <span class="font-extrabold text-amber-300">
                            Total: ${formatHM(monthCoreMins + (includeExtraDutiesInPrint ? monthExtraMins : 0))}
                            (Core: ${formatHM(monthCoreMins)}${includeExtraDutiesInPrint ? ` | Extra: ${formatHM(monthExtraMins)}` : ''})
                        </span>
                    </div>
                `;

                datesInMonth.forEach(dateStr => {
                    const log = appData[dateStr];
                    const dateObj = parseLocalDate(dateStr);
                    const dayFullStr = dateObj.toLocaleDateString('en-US', { weekday: 'long', month: 'short', day: 'numeric', year: 'numeric' });

                    const dayCard = document.createElement('div');
                    dayCard.className = "bg-white border border-slate-300 rounded-xl overflow-hidden avoid-break space-y-0";

                    dayCard.innerHTML = `
                        <div class="bg-slate-100 px-4 py-2 border-b border-slate-300 flex justify-between items-center">
                            <span class="font-bold text-xs text-slate-800 uppercase tracking-wide">${dayFullStr}</span>
                            ${log.homeroom ? `<span class="text-xs text-slate-600 font-medium">Duty: ${log.homeroom}</span>` : ''}
                        </div>
                    `;

                    const periodRows = (log.periods || []).filter(p => p.className || p.subject || p.topic).map(p => `
                        <tr class="border-b border-slate-200 text-xs">
                            <td class="py-2 px-3 font-bold text-slate-700 w-24">Period ${p.periodNum}</td>
                            <td class="py-2 px-3 font-semibold text-slate-800 w-28">${p.className || '-'} ${p.section || ''}</td>
                            <td class="py-2 px-3 text-slate-700 font-medium w-28">${p.workType || 'Teaching'}</td>
                            <td class="py-2 px-3 font-bold text-slate-900 w-36">${p.subject || '-'}</td>
                            <td class="py-2 px-3 text-slate-700">${p.topic || '-'} ${p.activity ? `<span class="text-slate-500 font-normal">(${p.activity})</span>` : ''}</td>
                        </tr>
                    `).join('');

                    const extraWorkRows = (log.extraWorks || []).map(w => `
                        <tr class="border-b border-amber-200 bg-amber-50/50 text-xs extra-duty-print-item">
                            <td class="py-2 px-3 font-bold text-amber-800"><i class="fa-solid fa-user-clock text-amber-600 mr-1"></i> Extra Duty</td>
                            <td class="py-2 px-3 font-semibold text-amber-900">${w.className || 'General'}</td>
                            <td class="py-2 px-3 text-amber-800 font-bold">${w.category || 'Special Duty'}</td>
                            <td class="py-2 px-3 font-bold text-amber-900">${w.durationMins || 0} Mins</td>
                            <td class="py-2 px-3 text-amber-900">${w.description || '-'}</td>
                        </tr>
                    `).join('');

                    dayCard.innerHTML += `
                        <table class="w-full text-left border-collapse">
                            <thead>
                                <tr class="bg-slate-50 text-[10px] text-slate-500 uppercase border-b border-slate-200">
                                    <th class="py-1.5 px-3">Slot</th>
                                    <th class="py-1.5 px-3">Class</th>
                                    <th class="py-1.5 px-3">Category</th>
                                    <th class="py-1.5 px-3">Subject</th>
                                    <th class="py-1.5 px-3">Topic / Work Content</th>
                                </tr>
                            </thead>
                            <tbody>
                                ${periodRows || '<tr><td colspan="5" class="p-3 text-xs text-slate-400 italic text-center">No core periods logged.</td></tr>'}
                                ${includeExtraDutiesInPrint ? extraWorkRows : ''}
                            </tbody>
                        </table>
                    `;

                    // RENDER BIG ATTACHED IMAGES FOR EXTRA WORKS IN PRINT MODE
                    if (includeExtraDutiesInPrint && (log.extraWorks || []).some(w => w.attachment)) {
                        const attachmentsHtml = (log.extraWorks || []).filter(w => w.attachment).map(w => `
                            <div class="print-attachment-container my-2 text-center">
                                <div class="text-[11px] font-bold text-slate-700 mb-1">
                                    Attachment for Extra Work (${w.category}): ${w.description || 'Document'}
                                </div>
                                <img src="${w.attachment}" class="print-extra-attachment mx-auto rounded-lg shadow-sm max-h-96" style="max-height: 480px; width: auto;" />
                            </div>
                        `).join('');

                        dayCard.innerHTML += `
                            <div class="p-3 bg-amber-50/30 border-t border-amber-200 extra-duty-print-item">
                                ${attachmentsHtml}
                            </div>
                        `;
                    }

                    if (log.dayNotes) {
                        dayCard.innerHTML += `
                            <div class="px-4 py-2 bg-slate-50 border-t border-slate-200 text-xs text-slate-600">
                                <strong class="text-slate-700">Day Notes:</strong> ${log.dayNotes}
                            </div>
                        `;
                    }

                    monthSection.appendChild(dayCard);
                });

                wrapper.appendChild(monthSection);
            });
        }

        function populateSearchDropdowns() {
            const classSelect = document.getElementById('search-class');
            const worktypeSelect = document.getElementById('search-worktype');
            if (classSelect) classSelect.innerHTML = '<option value="">All Classes</option>' + profileData.classes.map(c => `<option value="${c}">${c}</option>`).join('');
            if (worktypeSelect) worktypeSelect.innerHTML = '<option value="">All Categories</option>' + profileData.workTypes.map(w => `<option value="${w}">${w}</option>`).join('');
        }

        function executeSearch() {
            const keyword = document.getElementById('search-keyword').value.toLowerCase().trim();
            const classFilter = document.getElementById('search-class').value.toLowerCase().trim();
            const worktypeFilter = document.getElementById('search-worktype').value.toLowerCase().trim();
            const container = document.getElementById('search-results-list');
            if (!container) return;
            container.innerHTML = '';

            let matchCount = 0;
            Object.keys(appData).sort().reverse().forEach(date => {
                const log = appData[date];

                const matchingPeriods = (log.periods || []).filter(p => {
                    const matchKwd = !keyword || (p.topic && p.topic.toLowerCase().includes(keyword)) || (p.activity && p.activity.toLowerCase().includes(keyword));
                    const matchClass = !classFilter || (p.className && p.className.toLowerCase().includes(classFilter));
                    const matchWorktype = !worktypeFilter || (p.workType && p.workType.toLowerCase().includes(worktypeFilter));
                    return matchKwd && matchClass && matchWorktype && (p.className || p.subject || p.topic);
                });

                const matchingExtraWorks = (log.extraWorks || []).filter(w => {
                    const matchKwd = !keyword || (w.description && w.description.toLowerCase().includes(keyword));
                    const matchClass = !classFilter || (w.className && w.className.toLowerCase().includes(classFilter));
                    const matchWorktype = !worktypeFilter || (w.category && w.category.toLowerCase().includes(worktypeFilter));
                    return matchKwd && matchClass && matchWorktype;
                });

                if (matchingPeriods.length > 0 || matchingExtraWorks.length > 0) {
                    matchCount += (matchingPeriods.length + matchingExtraWorks.length);
                    const dateHeader = document.createElement('div');
                    dateHeader.className = "p-3 bg-slate-100 font-bold text-xs text-slate-700 flex justify-between items-center";
                    dateHeader.innerHTML = `<span><i class="fa-regular fa-calendar-days text-brand-600 mr-2"></i> ${date}</span><button onclick="jumpToDate('${date}')" class="text-brand-600 hover:underline no-print">Open Log &rrarr;</button>`;
                    container.appendChild(dateHeader);

                    matchingPeriods.forEach(p => {
                        const item = document.createElement('div');
                        item.className = "p-3 hover:bg-slate-50 border-b border-slate-100 text-xs";
                        item.innerHTML = `<p class="font-bold text-slate-800">Period ${p.periodNum}: ${p.subject} (${p.className || 'General'}) - <span class="text-brand-600">${p.workType || 'Teaching'}</span></p><p class="text-slate-600">Topic: ${p.topic || 'N/A'}</p>`;
                        container.appendChild(item);
                    });

                    matchingExtraWorks.forEach(w => {
                        const item = document.createElement('div');
                        item.className = "p-3 bg-amber-50/40 hover:bg-amber-50 border-b border-amber-100 text-xs space-y-2";
                        item.innerHTML = `
                            <div>
                                <p class="font-bold text-amber-900"><i class="fa-solid fa-user-clock mr-1"></i> Extra Work: ${w.category} (${w.durationMins} Mins)</p>
                                <p class="text-amber-800">${w.description || '-'}</p>
                            </div>
                            ${w.attachment ? `<div class="mt-2"><img src="${w.attachment}" class="max-h-64 rounded-md border border-amber-200" /></div>` : ''}
                        `;
                        container.appendChild(item);
                    });
                }
            });
            document.getElementById('search-count').innerText = `${matchCount} records found`;
        }

        function exportDataJSON() {
            const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify({ appData, profileData, timetableData }, null, 2));
            const dl = document.createElement('a');
            dl.setAttribute("href", dataStr);
            dl.setAttribute("download", `Teacher_Daybook_Backup_${getTodayDateStr()}.json`);
            document.body.appendChild(dl); dl.click(); dl.remove();
        }

        function exportDataCSV() {
            let csv = "data:text/csv;charset=utf-8,Date,Type,Period,Class,WorkType,Subject,Topic,DurationMins\n";
            Object.keys(appData).sort().forEach(date => {
                const log = appData[date];
                (log.periods || []).forEach(p => {
                    csv += `${date},Core,${p.periodNum},"${p.className || ''}","${p.workType || ''}","${p.subject || ''}","${p.topic || ''}",45\n`;
                });
                (log.extraWorks || []).forEach(w => {
                    csv += `${date},Extra,N/A,"${w.className || ''}","${w.category || ''}","Extra Duty","${w.description || ''}",${w.durationMins || 0}\n`;
                });
            });
            const link = document.createElement('a');
            link.setAttribute('href', encodeURI(csv));
            link.setAttribute('download', `Teacher_Daybook_${getTodayDateStr()}.csv`);
            document.body.appendChild(link); link.click(); link.remove();
        }

        function importDataJSON(e) {
            const file = e.target.files[0];
            if (!file) return;
            const r = new FileReader();
            r.onload = function(evt) {
                try {
                    const imported = JSON.parse(evt.target.result);
                    if (imported.appData) appData = imported.appData;
                    if (imported.profileData) profileData = imported.profileData;
                    if (imported.timetableData) timetableData = imported.timetableData;
                    saveLocalStorageData();
                    updateProfileUIHeadings();
                    renderProfileConfigLists(); renderPeriodRows(); loadDateLog(); renderTimetableBuilder(); populateSearchDropdowns(); onReportScopeTypeChange();
                    closeModal('modal-backup');
                    showToast("Backup imported successfully!");
                } catch (err) { alert("Invalid JSON backup file."); }
            };
            r.readAsText(file);
        }

        function switchTab(tabId) {
            currentTabId = tabId;
            ['planner', 'timetable', 'analytics', 'report', 'search'].forEach(t => {
                const view = document.getElementById(`view-${t}`);
                if (view) {
                    view.classList.add('hidden');
                    view.classList.remove('active-print-view');
                }
                const btn = document.getElementById(`tab-${t}`);
                if (btn) { btn.classList.remove('bg-brand-800', 'text-white'); btn.classList.add('text-brand-100'); }
                const mobBtn = document.getElementById(`mob-tab-${t}`);
                if (mobBtn) { mobBtn.classList.remove('text-white'); mobBtn.classList.add('text-brand-200'); }
            });

            const target = document.getElementById(`view-${tabId}`);
            if (target) {
                target.classList.remove('hidden');
                target.classList.add('active-print-view');
            }

            const activeBtn = document.getElementById(`tab-${tabId}`);
            if (activeBtn) { activeBtn.classList.add('bg-brand-800', 'text-white'); activeBtn.classList.remove('text-brand-100'); }

            const activeMobBtn = document.getElementById(`mob-tab-${tabId}`);
            if (activeMobBtn) { activeMobBtn.classList.add('text-white'); activeMobBtn.classList.remove('text-brand-200'); }

            if (tabId === 'timetable') renderTimetableBuilder();
            if (tabId === 'analytics') renderAnalytics();
            if (tabId === 'report') { onReportScopeTypeChange(); }
            if (tabId === 'search') executeSearch();
        }

        function showToast(msg) {
            const toast = document.getElementById('toast');
            document.getElementById('toast-msg').innerText = msg;
            toast.classList.remove('translate-y-20', 'opacity-0');
            toast.classList.add('translate-y-0', 'opacity-100');
            setTimeout(() => {
                toast.classList.remove('translate-y-0', 'opacity-100');
                toast.classList.add('translate-y-20', 'opacity-0');
            }, 2500);
        }
